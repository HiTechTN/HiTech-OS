  # NixOS branche hashedPasswordFile sur systemd LoadCredential=, qui

  # ("Failed to load environment files"). Conséquence : le "système
  # nerveux" (MQTT → Telegraf → InfluxDB) mort dès le premier boot.
  # MAINTENANT : systemd-tmpfiles crée un fichier placeholder (hash
  # sha512-crypt d'un mot de passe jeton "placeholder-change-me") UNIQUEMENT
  # s'il n'existe pas encore (la ligne tmpfiles `f` laisse intact tout
  # fichier déjà présent — le vrai secret injecté au déploiement par

    # hashedPasswordFile : "remove the username: prefix"). Avec le préfixe,
    # mosquitto recevait "user:user:hash" → décodage impossible → crash.
    # Généré via : mosquitto_passwd -b -c /tmp/f hitechos-node placeholder-change-me

    # garantit que l'option environmentFiles (type `absolute path` dans le
    # module NixOS — le prefixe "-" de systemd est interdit par le typage)
    # reste satisfiable. Comme ci-dessus : jamais écrasé si déjà présent.
    "f /run/secrets/telegraf-influx-token 0600 root root - INFLUX_TOKEN="

    # ("connection refused"/auth) sans broker joignable ET sans creds —
    # le champ mqtt_consumer pointe vers $MQTT_USER/$MQTT_PASSWORD via

    # mosquitto ci-dessus ; les vrais secrets (Phase 6) devront fournir
    # les deux fichiers de façon cohérente.
    # IMPORTANT : systemd EnvironmentFile attend UNE variable par ligne —
    # les deux paires sur une seule ligne faisaient de MQTT_USER une
    # variable unique mal formée ("hitechos-node MQTT_PASSWORD=...")
    # et laissait MQTT_PASSWORD indéfini : telegraf (strict env handling

    # L'échappement \n est interprété par systemd-tmpfiles (docs :

    environmentFiles = [

      "/run/secrets/telegraf-mqtt-creds" # fournit $MQTT_USER/$MQTT_PASSWORD
    ];
    extraConfig = {
      inputs.mqtt_consumer = [{
        servers = [ "tcp://127.0.0.1:1883" ];
        topics = [ "hitechos/+/+/+" ];
        data_format = "json";

  services.openssh.settings.KbdInteractiveAuthentication = false;

# HiTech-OS — provisionnement de l'écosystème d'outillage HiTechTN
# (services locaux + app2nix) sur le nœud patient-zero.
#
# Importé par ./installation.nix (donc patient-zero ET toute installation
# Calamares) — jamais par l'ISO live (iso.nix) ni par le profil headless

# nœud de production léger, retirer simplement l'import de ce fichier.
#
# Audit du 22/10/2026 : avant cet ajout, patient-zero ne contenait NI git
# NI docker (seule l'ISO live avait git — iso.nix), alors que :
# - La pile de services locaux exige : git · docker · docker compose ·
#   bash 4+ — elle s'installe via install.sh → docker compose up
#   (PostgreSQL+pgvector, MinIO, Jaeger, services applicatifs).
# - app2nix s'installe via Docker (`docker run ghcr.io/hitechtn/app2nix`),
#   via pip (Python), ou nativement via SON propre flake.nix (donc nix
#   flakes + nix-command — déjà activés dans ./common.nix).
# - bash 4+ et curl sont déjà dans la closure de base NixOS (vérifié via
#   `nix eval .#nixosConfigurations.patient-zero.config.environment.systemPackages`).

{ config, lib, pkgs, ... }:

{
  # --- Outils de base requis par les deux outils ---
  environment.systemPackages = with pkgs; [
    git # clone des dépôts de services/app2nix + usage de leurs scripts
    docker-compose # API `docker compose` (services locaux : install.sh)
    python3 # chemin "pip" natif d'app2nix (pyproject.toml/uv.lock)
    uv # gestion d'environnement Python d'app2nix (recommandé par son repo)
  ];

  # --- Conteneurs (les services locaux et app2nix utilisent Docker) ---
  virtualisation.docker.enable = true;

  # --- L'administrateur pilote Docker sans root (nécessaire pour
  # `docker compose up -d` des services locaux en tant que hitechos) ---
  users.users.hitechos.extraGroups = [ "docker" ];

  # NOTE Sécurité (à documenter lors du déploiement) : Docker publie ses
  # ports via ses propres chaînes iptables/nftables, ce qui peut contourner
  # le firewall NixOS (networking.firewall dans ./common.nix) pour les
  # ports publiés par le démon Docker. Restreindre les publications de
  # ports aux interfaces de confiance si le nœud est exposé.
}

--- FILE: ./os/netinstaller.nix ---
# HiTech-OS — Phase 8 NetInstaller (minimal eframe/egui ISO).
#
# Goal: a small (~800 MB target) live ISO that contains only:
#   - a Wayland kiosk running the Rust `hitech-installer` GUI,
#   - networking tools,
#   - git + curl,
#   - the NixOS installer machinery.
#
# The target closure is NOT in the ISO, but the complete HiTech-OS source tree
# IS embedded at /etc/hitech-os-source. The graphical installer:
#   1. partitions/formats/mounts the target,
#   2. copies the embedded source tree offline-first (git is only a fallback),
#   3. writes the operator's anonymous identity into configuration.nix,
#   4. calls `nixos-install` with the HiTech-Store binary cache as the primary
#      substituter, so the target closure is downloaded instead of rebuilt.
#
# Two outputs are therefore produced by the flake:
#   nixosConfigurations.netinstaller       — this minimal ISO
#   nixosConfigurations.iso                — legacy full graphical ISO
#
# Security contract:
#   - `cachePublicKey` is embedded in the ISO; the cache is trusted only if
#     its signature verifies. Leave it empty in development only.
#   - `allowSourceFallback = false` is the production default once the cache
#     has a stable release; it prevents a compromised/down cache from silently
#     rebuilding unvetted source on the target.

{ config, lib, pkgs, self, hitechosRevision ? "main", ... }:

let
  cfg = config.hitechos.netinstaller;

  # Pull the HiTech-Store/Harmonia cache public key from the option. The NixOS
  # module for the cache server exposes the same value as a read-only option
  # when co-located; this stays a plain string in the ISO so the installer has
  # no dependency on the server module.
  cachePublicKey = cfg.cachePublicKey;

  # Channel-free installer: the ISO does not embed the full Nixpkgs source
  # tree (that alone is hundreds of MB). Instead the live script points
  # NIX_PATH at the exact Nixpkgs revision baked into the ISO; nixos-install
  # fetches it over the network, which the NetInstaller requires anyway for the
  # closure cache. The revision is the same one used to build the ISO, so the
  # installed system stays reproducible.
  nixpkgsArchive =
    if config.system.nixos.revision != null then
      "https://github.com/NixOS/nixpkgs/archive/${config.system.nixos.revision}.tar.gz"
    else
      "https://github.com/NixOS/nixpkgs/archive/nixos-unstable.tar.gz";

  # Graphical HiTech-OS installer (eframe/egui). It replaces Calamares and
  # owns the partition/format/mount/install pipeline.
  hitechInstaller = pkgs.callPackage ../hitech-installer { };

  # Wrapper script that detects missing hardware acceleration and forces
  # software rendering (llvmpipe) so the eframe/glow HUD appears instead of
  # a black screen on VirtualBox/QEMU-without-virgl/headless CI.
  #
  # Detection strategy:
  #  1. If NO DRM render node exists at all (not just renderD0), force
  #     software rendering for both GL and Wayland/Ozone.
  #  2. Check for common virtual-machine indicators (QEMU virtio-gpu, VirtualBox
  #     VBoxGuest, VMware vmware) so llvmpipe is used even when a stub DRM
  #     device is present but hardware acceleration is not usable.
  hitechInstallerKiosk = pkgs.writeScriptBin "hitech-installer-kiosk" ''
    soft=0
    if [ ! -e /dev/dri/renderD0 ] && [ ! -e /dev/dri/card0 ]; then
      soft=1
    fi
    if [ -e /sys/module/virtio_gpu ] || [ -e /sys/module/vboxguest ] || grep -aq vmware /proc/1/environ 2>/dev/null; then
      soft=1
    fi
    if [ "$soft" = "1" ]; then
      export LIBGL_ALWAYS_SOFTWARE=1
      export WLR_RENDERER_ALLOW_SOFTWARE=1
      export WLR_RENDERER=soft
      export EGL_PLATFORM=x11
      export GST_VAAPI_ALL_DRIVERS=1
    fi
    exec ${hitechInstaller}/bin/hitech-installer "$@"
  '';

in
{
  imports = [ ./hardware-universal.nix ];

  options.hitechos.netinstaller = {
    enable = lib.mkOption {
      type = lib.types.bool;
      default = true;
      description = ''
        Enable the Phase 8 NetInstaller ISO configuration. This module is only
        imported by `nixosConfigurations.netinstaller`, so it defaults to true;
        set false only to evaluate the bare upstream minimal ISO.
      '';
    };

    repoUrl = lib.mkOption {
      type = lib.types.str;
      default = "https://github.com/HiTechTN/HiTech-OS.git";
      description = "Git remote fetched by the graphical installer inside the live session.";
    };

    repoRef = lib.mkOption {
      type = lib.types.str;
      default = hitechosRevision;
      defaultText = lib.literalExpression "self.rev or \"main\"";
      description = ''

# HiTech-OS — installeur "one-click" : greffe une étape supplémentaire
# sur l'installeur graphique Calamares officiel de NixOS, pour que
# l'installation sur disque inclue AUSSI nos services (common.nix), pas
# juste un NixOS générique.
#
# HISTORIQUE / POURQUOI CETTE VERSION (v2) :
# La première version patchait settings.conf avec `sed` (a\ + \n) dans le
# postInstall du dérivé. Testé en réel le 20/09/2026 sur du matériel
# physique (voir la conversation) : le sed n'a PRIS AUCUN EFFET dans le
# bac à sable de build Nix (aucune erreur, mais notre job "shellprocess@
# hitechos" n'apparaissait tout simplement pas dans la séquence exécutée
# par Calamares — silencieusement ignoré). Cause probable : GNU sed
# interprète \n dans le texte d'une commande `a\` différemment selon le
# contexte d'exécution, fragile à travers les sandboxes de build.
#
# Cette version (v2) élimine complètement le sed : le settings.conf
# complet est écrit tel quel en Nix (chaîne littérale ci-dessous),
# construit à partir du contenu RÉEL de calamares-nixos-extensions au
# commit nixpkgs épinglé dans flake.lock (20b1ddd1aa5ace70c9468305030aa
# 4f9ef79671b — la révision exacte qui a servi à builder l'ISO testée le
# 20/09/2026), avec seulement les deux ajouts nécessaires (une instance
# + une entrée de séquence). Zéro ambiguïté d'échappement shell : soit
# ça compile avec le bon contenu, soit `nix build` échoue directement.
#
# TRANSPARENCE : voir aussi le premier test réel — l'erreur rencontrée
# ("experimental Nix feature flakes is disabled") venait du job STOCK
# "nixos" (nixos-install), pas de ce fichier : corrigée séparément dans
# common.nix (nix.settings.experimental-features). Cette étape
# shellprocess elle-même n'a PAS ENCORE été testée en conditions réelles
# (le premier test a échoué avant d'atteindre ce point de la séquence,
# et de toute façon la version sed ne s'exécutait pas). Prochain test
# réel à surveiller.
#
# Mécanisme (une fois que ça s'exécute) :
# 1. Le job "nixos" (natif Calamares) partitionne, installe un NixOS de
#    base et écrit /etc/nixos/{configuration,hardware-configuration}.nix
#    — on ne touche PAS à ça, ce sont les réglages propres à CE disque
#    précis (bootloader EFI/BIOS, UUIDs de partitions...).
# 2. Notre étape "shellprocess" s'exécute juste après, chrootée dans le
#    système fraîchement installé (dontChroot: false) :
#    a. clone HiTech-OS dans /etc/nixos/hitech-os
#    b. insère l'import de os/common.nix (services, durcissement — AUCUN
#       réglage disque/bootloader dedans, donc pas de conflit avec ce que
#       "nixos" vient d'écrire) juste après la ligne
#       "./hardware-configuration.nix" du configuration.nix généré
#    c. `nixos-rebuild boot` pour que ce soit actif dès le premier
#       démarrage réel (pas besoin de "switch" : le système n'est pas
#       encore démarré, on ne fait que fixer la génération de boot par
#       défaut)

{ pkgs, ... }:

let
  # nixos-install retry wrapper used by both the stock Calamares job (through
  # the patched main.py) and our post-install shellprocess. This is the
  # defence against transient squashfs I/O errors and cache/network flakes.
  hitechosNixosInstall = pkgs.writeShellScriptBin "hitechos-nixos-install" ''
    set -u
    attempt=1
    status=1
    while [ "$attempt" -le 3 ]; do
      echo "hitechos: nixos-install attempt $attempt/3" >&2
      ${pkgs.nixos-install}/bin/nixos-install \
        --option fallback true \
        --option substitute true \
        "$@"
      status=$?
      if [ "$status" -eq 0 ]; then
        exit 0
      fi
      echo "hitechos: nixos-install failed (exit $status); retrying after 4s" >&2
      attempt=$((attempt + 1))
      sleep 4
    done
    exit "$status"
  '';

  # Orion appears alongside the stock desktop choices. The stock NixOS job
  # learns about the id in calamares-main.patch.py.
  hitechosPackagechooserConf = pkgs.writeText "packagechooser.conf" ''

    method: legacy

        step: "Desktop"

        - id: plasma6
          packages: [ plasma6 ]
          name: KDE Plasma 6
          description: "<html>Plasma is a highly customizable, modern Qt workspace that stays out of the way.<br/><br/>Learn more at <a href='https://kde.org/plasma-desktop/'>kde.org/plasma-desktop</a></html>"
          screenshot: "images/plasma6.jpg"

        - id: xfce
          packages: [ xfce ]
          name: XFCE
          description: "<html>Xfce is a lightweight desktop environment that aims to be fast and low on system resources.<br/><br/>Learn more at <a href='https://www.xfce.org/'>xfce.org</a></html>"
          screenshot: "images/xfce.jpg"
  '';  # Script exécuté dans le LIVE (dontChroot: true) avec le point de montage
  # de la cible en argument.
  #
  # Phase 8.3 (2026-10-28) : le job Calamares "nixos" (patché dans
  # calamares-main.patch.py) copie désormais le dépôt HiTech-OS bundleisé
  # depuis /etc/hitech-os-source (ISO) vers $ROOT/etc/nixos/hitech-os et
  # injecte l'import de os/installation.nix dans configuration.nix AVANT
  # d'appeler nixos-install. Ce script shellprocess s'exécute donc AFTER une

  # corrige le hostname si nécessaire, et force nixos-rebuild boot pour que
  # les modules HiTech-OS soient actifs au premier démarrage réel.
  #
  # Si pour une raison quelconque le bundle n'a PAS été copié (ISO ancienne

  hitechosPostInstall = pkgs.writeShellScript "hitechos-post-install" ''
    set -eu

    target="$1"
    repo_rel="/etc/nixos/hitech-os"
    repo="$target$repo_rel"

    cfg="$target/etc/nixos/configuration.nix"
    import_rel="./hitech-os/os/installation.nix"

    # UDisks2/GVFS must not keep a target partition busy while Calamares is
    # mounting it. Only live media paths are touched; never the live root.
    for mount in /run/media/* /media/*; do
      if [ -d "$mount" ] && mountpoint -q "$mount"; then
        umount -l "$mount" || true
      fi
    done

    # Phase 8.3 : le bundle /etc/hitech-os-source est fourni par l'ISO.
    # Si le job "nixos" a déjà fait lecopier, on n'a rien à faire.
    if [ -d "$repo" ] && [ -f "$repo/os/installation.nix" ] && \
       grep -qF "$import_rel" "$cfg" 2>/dev/null; then
      echo "hitechos: hitech-os already bundled + imported by nixos job"
    else
      # Ancidien chemin : clone git + injection manuelle (ISO sans bundle).
      if [ ! -d "$repo/.git" ]; then
        ${pkgs.git}/bin/git clone --depth 1 \
          https://github.com/HiTechTN/HiTech-OS.git "$repo"
      fi

      if [ ! -f "$repo/os/installation.nix" ]; then
        echo "FATAL: hitech-os/os/installation.nix not found; aborting" >&2
        exit 1
      fi

      if ! grep -qF "$import_rel" "$cfg"; then
        ${pkgs.gnused}/bin/sed -i "\|hardware-configuration.nix|a\\      $import_rel" "$cfg"
      fi

      # Ensure the bundled copy landed as real files, not symlink stubs
      # pointing back into the ISO /nix/store. Dereferencing avoids
      # chmod/read errors on the target when the closure is later
      # manipulated or when nixos-install traverses the tree.
      if [ -d "$repo/.git" ]; then
        rm -rf "$repo/.git"
      fi
    fi

    # that as the target hostname. Prefer the operator's chosen hostname and
    # fall back to hitechos only when the placeholder survived.
    if grep -q 'hitech-tobefilledbyoem' "$cfg"; then
      ${pkgs.gnused}/bin/sed -i 's/hitech-tobefilledbyoem/hitechos/' "$cfg"
    fi

    # If nixos-install already succeeded (bundle path), just ensure the
    # generation is the default boot entry. Otherwise run the full installer.
    if [ -d "$repo" ] && [ -f "$repo/os/installation.nix" ] && \

      ${pkgs.nixos-install}/bin/nixos-install \
        --root "$target" \
        --no-root-passwd \
        --upgrade \
        --option fallback true \
        --option substitute true \
        || true
    else
      # The wrapper adds --option fallback true / substitute true and retries
      # transient target I/O failures up to three times.
      ${hitechosNixosInstall}/bin/hitechos-nixos-install \
        --root "$target" \
        --no-root-passwd
    fi
  '';

  hitechosShellprocessConf = pkgs.writeText "shellprocess-hitechos.conf" ''

    # reconstruit la cible sans dépendre de son git/nixos-rebuild.
    #
    # Correctif Phase 7 : l'ancienne version tournait chrootée (dontChroot:
    # false) et supposait git présent dans la base NixOS fraîchement
    # installée. git n'y est pas — seul l'ISO l'avait — d'où l'échec réel
    # "git: command not found" et l'arrêt de toute la séquence Calamares.
    dontChroot: true
    timeout: 3600
    script:
      - '${hitechosPostInstall} "''${ROOT}"'
  '';

    # to its module name, e.g. "welcome@welcome". In the *sequence* section,
    # mentioning a module without a full instance key (e.g. "welcome")
    # means that implicit module.

    # module contains an example `welcome.conf`. Specify a *config* for
    # any module (also implicit instances) to change which file is used.
    #
    # For more information on running module instances, run Calamares in debug
    # mode and check the Modules page in the Debug information interface.
    #
    # A module that is often used with instances is shellprocess, which will
    # run shell commands specified in the configuration file. By configuring
    # more than one instance of the module, multiple shell sessions can be run
    # during install.

    # viewmodules and jobmodules, as they should appear and/or run.

    # a viewmodule instance key (or name) can appear in both exec and show phases.
    # There is no limit to the number of show or exec phases. However, the same
    # module instance key should not appear more than once per phase, and
    # deployers should take notice that the global storage structure is persistent
    # throughout the application lifetime, possibly influencing behavior across
    # phases. A show phase defines a sequence of viewmodules (and therefore
    # pages). These viewmodules can offer up jobs for the execution queue.

    # progress page iterates over the modules listed in the *immediately
    # preceding* show phase, and enqueues their jobs, as well as any other jobs
    # from jobmodules, in the order defined in the current exec phase.
    #
    # It then executes the job queue and clears it. If a viewmodule offers up a
    # job for execution, but the module name (or instance key) isn't listed in the
    # immediately following exec phase, this job will not be executed.

    "iwlmvm"

    "rtw89_core"

# HiTech-OS — binary cache for the NetInstaller and OTA closures.
#
# Design rationale (see docs/audit/PHASE8-NETINSTALLER-RUST-DE.md):
#   - Harmonia is the primary cache server: Rust, low overhead, multi-worker,
#     and it signs store paths with a dedicated key.
#   - nix-serve is documented as a fallback if a host already uses it, but it
#     does not sign and is single-process; this module intentionally uses
#     Harmonia for the new cache.
#   - The cache is write-protected and read-only to clients. CI never gets a
#     private key in a writable repository: it builds closures and pushes them
#     to the cache host over SSH; the cache host signs them.
#   - TLS terminates in nginx with ACME. Plain HTTP is only for the loopback
#     listener.
#
# Deployment contract:
#   /run/secrets/hitech-cache-signing-key — secret key generated offline with
#       nix-store --generate-binary-cache-key hitechos-cache-1 <secret> <public>
#   services.hitech-cache.publicKey       — the matching public key, embedded
#       in clients and in the NetInstaller ISO.
#
# IMPORTANT: keep the private key out of Git. The public key alone is safe to
# commit and must be embedded in the installer so cache signatures can be
# verified.

{ config, lib, pkgs, ... }:

let
  cfg = config.services.hitech-cache;
in
{
  options.services.hitech-cache = {
    enable = lib.mkEnableOption "Harmonia binary cache server for HiTech-OS closures";

    package = lib.mkPackageOption pkgs "harmonia" { };

    bindAddress = lib.mkOption {
      type = lib.types.str;
      default = "127.0.0.1";
      description = "Address Harmonia binds to; nginx proxies external traffic.";
    };

    port = lib.mkOption {
      type = lib.types.port;
      default = 5000;
      description = "Internal Harmonia TCP port.";
    };

    domain = lib.mkOption {
      type = lib.types.str;
      default = "cache.hitechos.local";
      description = "Public hostname served by nginx (TLS/ACME).";
    };

    enableTls = lib.mkOption {
      type = lib.types.bool;
      default = false;
      description = ''
        Terminate TLS with nginx + ACME. Enable once `domain` resolves to this
        host and port 80/443 are reachable from the internet. Disable only for
        a private network/VPN cache.
      '';
    };

    signKeyPath = lib.mkOption {
      type = lib.types.path;
      default = "/run/secrets/hitech-cache-signing-key";
      description = ''
        Path to the secret signing key. Provision it with sops/agenix, never
        with systemd-tmpfiles and never in the Nix store.
      '';
    };

    publicKey = lib.mkOption {
      type = lib.types.str;
      default = "";
      example = "hitechos-cache-1:AbCdEf...=";
      description = ''
        Public key corresponding to `signKeyPath`. Required when the server is
        enabled; clients must trust this key to accept HiTech-OS closures.
      '';
    };

    openFirewall = lib.mkOption {
      type = lib.types.bool;
      default = false;
      description = "Open the public cache ports in the NixOS firewall.";
    };

    client = {
      enable = lib.mkOption {
        type = lib.types.bool;
        default = false;
        description = ''
          Configure this host to substitute from the HiTech-OS cache. Enable on
          patient-zero, installed targets and the NetInstaller live system.
        '';
      };

      url = lib.mkOption {
        type = lib.types.str;
        default = "https://cache.hitechos.local";
        description = "Base URL of the HiTech-OS binary cache.";
      };

      publicKey = lib.mkOption {
        type = lib.types.str;
        default = "";
        description = "Trusted public key for `url` (must match the server).";
      };
    };
  };

  config = lib.mkMerge [
    (lib.mkIf cfg.enable {
      assertions = [
        {
          assertion = cfg.publicKey != "";

            services.hitech-cache.enable requires services.hitech-cache.publicKey
            so clients can verify cache signatures.
          '';
        }
      ];

      services.harmonia = {
        inherit (cfg) package;

        cache = {
          enable = true;
          signKeyPaths = [ cfg.signKeyPath ];
          settings.bind = "${cfg.bindAddress}:${toString cfg.port}";
        };
      };

      services.nginx = lib.mkIf cfg.enableTls {
        enable = true;
        recommendedProxySettings = true;
        recommendedTlsSettings = true;

        virtualHosts.${cfg.domain} = {
          enableACME = true;
          forceSSL = true;
          locations."/".proxyPass = "http://${cfg.bindAddress}:${toString cfg.port}";
        };
      };

      security.acme.acceptTerms = lib.mkIf cfg.enableTls true;

# HiTech-OS — service natif « hitech-store » (portail universel FastAPI)
#
# Packagé via la derivation « à la app2nix » : apps/hitech-store/default.nix.
# Importé par os/installation.nix, donc présent sur patient-zero ET sur
# toute installation Calamares (active par défaut, désactivable).
#
# Secrets : /run/secrets/hitech-store.env est un placeholder créé par
# systemd-tmpfiles UNIQUEMENT s'il n'existe pas. os/services/secrets.nix le
# remplace avant le démarrage si /etc/hitechos/hitech-store.env.enc existe.
# Le panel admin reste utilisable sans SMTP en mode DEV_OTP_FALLBACK=1
# (PIN écrit dans le journal) ; passer devOtpFallback=false en production.

{ config, lib, pkgs, ... }:

let
  cfg = config.services.hitech-store;
  hitechStore = pkgs.callPackage ../../apps/hitech-store/default.nix { };

  # The tmpfiles placeholder ships a publicly known SECRET_KEY so that the
  # service can boot before sops secrets are provisioned. That key signs the
  # admin JWT: if it survives into an exposed deployment, anyone can forge an
  # admin session. This oneshot replaces it with a per-boot random key when the
  # placeholder (or an empty value) is still present; a real sops-provisioned
  # key is never touched. Runs after hitechos-secrets, before the service.
  envKeygen = pkgs.writeShellScript "hitech-store-keygen" ''
    set -eu
    umask 077
    env_file=${lib.escapeShellArg cfg.environmentFile}
    [ -f "$env_file" ] || exit 0

    placeholder='^SECRET_KEY=dev-only-placeholder-change-me$'
    if ${pkgs.gnugrep}/bin/grep -q "$placeholder" "$env_file" \
      || ! ${pkgs.gnugrep}/bin/grep -q '^SECRET_KEY=.' "$env_file"; then
      new_key="$(${pkgs.openssl}/bin/openssl rand -hex 32)"
      tmp="$env_file.tmp.$$"
      trap '${pkgs.coreutils}/bin/rm -f "$tmp"' EXIT
      ${pkgs.gawk}/bin/awk -v key="$new_key" '
        BEGIN { replaced = 0 }
        /^SECRET_KEY=/ { print "SECRET_KEY=" key; replaced = 1; next }
        { print }
        END { if (!replaced) print "SECRET_KEY=" key }
      ' "$env_file" > "$tmp"
      ${pkgs.coreutils}/bin/chmod --reference="$env_file" "$tmp" 2>/dev/null || true
      ${pkgs.coreutils}/bin/chown --reference="$env_file" "$tmp" 2>/dev/null || true
      ${pkgs.coreutils}/bin/mv -f "$tmp" "$env_file"
      trap - EXIT
      echo "hitech-store-keygen: generated a per-boot SECRET_KEY (placeholder was still present)" >&2
    fi
  '';
in

# HiTech-OS — déchiffrement sops des secrets de service au boot.
#
# Pourquoi un oneshot et pas sops-nix comme module NixOS ?
# L'installateur Calamares génère un /etc/nixos/configuration.nix NON-flake.
# Un `imports = [ sops-nix.nixosModules.sops ]` y serait impossible sans
# flake/input disponible au moment du `nixos-rebuild` local. On garde donc
# le contrat NixOS simple : des fichiers chiffrés déposés hors dépôt, et
# sops (paquet) appelé au démarrage.
#
# Contrat de déploiement (répertoire root:root 0700) :
#   /etc/hitechos/age.key                  clé age de l'hôte (jamais commitée)
#   /etc/hitechos/hitech-store.env.enc     secrets du portail (format dotenv)
#   /etc/hitechos/genio.env.enc            tokens store/OpenRouter de Genio
#
# Création (exemple) :
#   sops --encrypt --age "$(cat /etc/hitechos/age.pub)" \
#        --input-type dotenv --output-type dotenv \
#        /etc/hitechos/hitech-store.env > /etc/hitechos/hitech-store.env.enc
#
# Sans fichiers .enc, le service ne fait rien : les placeholders tmpfiles
# restent en place et les services démarrent en mode dégradé explicite.

{ config, lib, pkgs, ... }:

let
  secretsDir = "/etc/hitechos";
  decrypt = pkgs.writeShellScript "hitechos-decrypt-secrets" ''
    set -eu
    umask 077
    ${pkgs.coreutils}/bin/install -d -m 0700 /run/secrets

    group_exists() {
      while IFS=: read -r name _; do
        [ "$name" = "$1" ] && return 0
      done < /etc/group
      return 1
    }

    decrypt_one() {
      src="$1"
      dst="$2"
      group="$3"
      [ -f "$src" ] || return 0
      if [ ! -r "${secretsDir}/age.key" ]; then
        echo "hitechos-secrets: $src exists but ${secretsDir}/age.key is missing" >&2
        return 1
      fi
      if ! group_exists "$group"; then
        echo "hitechos-secrets: group $group absent, skipping $dst" >&2
        return 0
      fi
      tmp="$(${pkgs.coreutils}/bin/mktemp)"
      trap '${pkgs.coreutils}/bin/rm -f "$tmp"' EXIT
      SOPS_AGE_KEY_FILE="${secretsDir}/age.key" \
        ${pkgs.sops}/bin/sops --decrypt "$src" > "$tmp"
      ${pkgs.coreutils}/bin/install -o root -g "$group" -m 0640 "$tmp" "$dst"
      ${pkgs.coreutils}/bin/rm -f "$tmp"
      trap - EXIT
    }

    decrypt_one "${secretsDir}/hitech-store.env.enc" /run/secrets/hitech-store.env hitech-store
    decrypt_one "${secretsDir}/genio.env.enc" /run/secrets/genio.env genio
  '';
in
{
  environment.etc."hitechos/README".text = ''
    HiTech-OS secrets
    =================
    Place the age private key in /etc/hitechos/age.key (mode 0600).
    Encrypt dotenv files with sops and name them:
      hitech-store.env.enc
      genio.env.enc
    They are decrypted to /run/secrets/ before the services start.
    Required hitech-store keys: SECRET_KEY, SMTP_HOST, SMTP_PORT, SMTP_USER,
      SMTP_PASSWORD, AGENT_SHARED_SECRET, STORAGE_DIR.
    Optional GitHub OAuth keys: GITHUB_CLIENT_ID, GITHUB_CLIENT_SECRET,
      GITHUB_ALLOWED_LOGIN, GITHUB_REDIRECT_URI.
    Required genio keys: HITECH_STORE_ADMIN_TOKEN, HITECH_STORE_AGENT_TOKEN;
      optional GENIO_OPENROUTER_KEY / OPENROUTER_API_KEY.
  '';

  systemd.services.hitechos-secrets = {
    description = "Decrypt HiTech-OS service secrets with sops";
    wantedBy = [ "multi-user.target" ];
    before = [ "hitech-store.service" "genio.service" ];
    serviceConfig = {
      Type = "oneshot";
      RemainAfterExit = true;
      ExecStart = "${decrypt}";
    };
  };
}

#!/usr/bin/env bash
#
# HiTech-OS — sign a store path and its closure with the cache signing key.
#
# Runs on the cache host (or any trusted signer that owns the private key).
# Intended to be called over SSH by scripts/publish-cache.sh and the
# .github/workflows/cache-publish.yml pipeline; it can also be used manually
# after a closure has been copied into the cache host's store.
#
# Usage:
#   scripts/sign-cache-closure.sh [-k KEY] PATH [PATH ...]
#
# Options:
#   -k, --key PATH   secret signing key (default: $HITECH_CACHE_SIGNING_KEY or
#                    /run/secrets/hitechos-cache-signing-key)
#   -h, --help       show this help
#
# Environment:
#   HITECH_CACHE_SIGNING_KEY  default key path
#
# Example (on the cache host):
#   scripts/sign-cache-closure.sh /nix/store/<hash>-nixos-system-patient-zero-<ver>

set -euo pipefail

NIX=(nix --extra-experimental-features "nix-command flakes")

die() { printf 'sign-cache-closure: error: %s\n' "$*" >&2; exit 1; }
log() { printf 'sign-cache-closure: %s\n' "$*" >&2; }

usage() { sed -n '2,22p' "$0" | sed 's/^# \{0,1\}//'; }

key=${HITECH_CACHE_SIGNING_KEY:-/run/secrets/hitechos-cache-signing-key}
paths=()

while [ $# -gt 0 ]; do
  case "$1" in
    -k|--key) key=$2; shift 2 ;;
    -h|--help) usage; exit 0 ;;
    --) shift; while [ $# -gt 0 ]; do paths+=("$1"); shift; done ;;
    -*) die "unknown option: $1" ;;
    *) paths+=("$1"); shift ;;
  esac
done

[ "${#paths[@]}" -gt 0 ] || { usage; exit 2; }
[ -r "$key" ] || die "signing key is not readable: $key"
command -v nix >/dev/null || die "nix is not in PATH"

for path in "${paths[@]}"; do
  [ -e "$path" ] || die "store path does not exist: $path"
  log "signing closure of $path"
  # Sign every closure member: Nix verifies each path it substitutes.
  "${NIX[@]}" path-info --recursive "$path" | "${NIX[@]}" store sign --key-file "$key" --stdin

--- FILE: ./scripts/publish-cache.sh ---
#!/usr/bin/env bash
#
# HiTech-OS — publish system closures to the Harmonia binary cache.
#
# Builds one or more flake targets, optionally signs them with a local secret
# key, uploads the closure to the cache host over SSH, and optionally lets the
# cache host sign the closure with the key that lives there
# (default: /run/secrets/hitechos-cache-signing-key).
#
# Security model (see docs/audit/PHASE8-NETINSTALLER-RUST-DE.md §5):
#   - The cache host owns the private signing key. CI never receives it.
#   - CI copies closures over a restricted SSH key, then the cache host signs
#     them. Clients verify the signatures with the matching public key.
#   - A trusted builder that does have the key can sign locally with --sign-key.
#
# Usage:
#   scripts/publish-cache.sh [options] [target ...]
#
# Targets:
#   patient-zero   nixosConfigurations.patient-zero system closure (default)
#   hitech-shell   Rust DE package (default)
#   netinstaller   NetInstaller ISO
#   iso            legacy full ISO
#   <installable>  any flake installable passed through to `nix build`
#
# Options:
#   --repo PATH          repository root (default: git toplevel)
#   --host user@host     cache host SSH target (or HITECH_CACHE_HOST)
#   --sign-key PATH      local secret key; signs before upload
#   --remote-key PATH    key path on the cache host; signs after upload
#   --targets "a b"      space-separated targets (default: patient-zero hitech-shell)
#   --no-push            build (and optionally sign) locally, skip the network
#   --no-sign            never sign, even if a key was supplied
#   --dry-run            print what would be built/pushed
#   -h, --help           show this help
#
# Environment:
#   HITECH_CACHE_HOST         cache host SSH target ("user@host")
#   HITECH_CACHE_SIGNING_KEY  local secret key (trusted builders only)
#   HITECH_CACHE_REMOTE_KEY   key path on the cache host
#   HITECH_CACHE_TARGETS      space-separated default targets
#   HITECH_REPO_ROOT          override repository root
#
# Examples:
#   # trusted builder with the key mounted locally
#   scripts/publish-cache.sh --sign-key /secure/hitechos-cache-1.secret
#
#   # CI: push unsigned, let the cache host sign
#   HITECH_CACHE_HOST=cache-publisher@cache.hitechos.tn \
#     scripts/publish-cache.sh
#
#   # build and sign locally, no network
#   scripts/publish-cache.sh --no-push --sign-key /secure/hitechos-cache-1.secret

set -euo pipefail

NIX=(nix --extra-experimental-features "nix-command flakes")

die() { printf 'publish-cache: error: %s\n' "$*" >&2; exit 1; }
log() { printf 'publish-cache: %s\n' "$*" >&2; }

usage() { sed -n '2,53p' "$0" | sed 's/^# \{0,1\}//'; }

repo_root() {
  if [ -n "${HITECH_REPO_ROOT:-}" ]; then
    printf '%s' "$HITECH_REPO_ROOT"
    return
  fi
  if root=$(git -C "$(dirname "$0")" rev-parse --show-toplevel 2>/dev/null); then
    printf '%s' "$root"
    return
  fi
  printf '%s' "$(cd "$(dirname "$0")/.." && pwd)"
}

target_attr() {
  case "$1" in
    patient-zero) printf '%s' '.#nixosConfigurations.patient-zero.config.system.build.toplevel' ;;
    hitech-shell) printf '%s' '.#hitech-shell' ;;
    netinstaller) printf '%s' '.#nixosConfigurations.netinstaller.config.system.build.isoImage' ;;
    iso) printf '%s' '.#nixosConfigurations.iso.config.system.build.isoImage' ;;
    *) printf '%s' "$1" ;;
  esac
}

# Sign every path in the closure. Nix verifies each substituted path, so every
# closure member needs a signature, not only the toplevel.
sign_closure() {
  local key=$1 path=$2
  [ -r "$key" ] || die "signing key is not readable: $key"
  log "signing closure of $path with $key"
  "${NIX[@]}" path-info --recursive "$path" | "${NIX[@]}" store sign --key-file "$key" --stdin
}

push_closure() {
  local host=$1 path=$2
  log "pushing $path -> ssh-ng://$host"
  "${NIX[@]}" copy --to "ssh-ng://$host" --no-check-sigs "$path"
}

sign_closure_remote() {
  local host=$1 key=$2 path=$3
  log "signing $path on $host with $key"
  ssh -o BatchMode=yes -o ConnectTimeout=15 "$host" \
    "nix --extra-experimental-features 'nix-command flakes' path-info --recursive '$path' | nix --extra-experimental-features 'nix-command flakes' store sign --key-file '$key' --stdin"
}

verify_remote() {
  local host=$1 path=$2
  "${NIX[@]}" path-info --store "ssh-ng://$host" "$path" >/dev/null
  log "verified present in cache: $path"
}

main() {
  local repo="" host="" sign_key="" remote_key="" no_push=0 no_sign=0 dry_run=0
  local -a targets=()

  while [ $# -gt 0 ]; do
    case "$1" in
      --repo) repo=$2; shift 2 ;;
      --host) host=$2; shift 2 ;;
      --sign-key) sign_key=$2; shift 2 ;;
      --remote-key) remote_key=$2; shift 2 ;;
      --targets) read -r -a targets <<< "$2"; shift 2 ;;
      --no-push) no_push=1; shift ;;
      --no-sign) no_sign=1; shift ;;
      --dry-run) dry_run=1; shift ;;
      -h|--help) usage; exit 0 ;;
      --) shift; while [ $# -gt 0 ]; do targets+=("$1"); shift; done ;;
      -*) die "unknown option: $1" ;;
      *) targets+=("$1"); shift ;;
    esac
  done

  repo=${repo:-$(repo_root)}
  host=${host:-${HITECH_CACHE_HOST:-}}
  sign_key=${sign_key:-${HITECH_CACHE_SIGNING_KEY:-}}
  remote_key=${remote_key:-${HITECH_CACHE_REMOTE_KEY:-/run/secrets/hitechos-cache-signing-key}}
  host=${host#ssh-ng://}

  if [ "${#targets[@]}" -eq 0 ]; then
    read -r -a targets <<< "${HITECH_CACHE_TARGETS:-patient-zero hitech-shell}"
  fi

  [ -d "$repo" ] || die "repository root not found: $repo"
  command -v nix >/dev/null || die "nix is not in PATH"

  if [ "$no_push" -eq 0 ] && [ "$dry_run" -eq 0 ] && [ -z "$host" ]; then
    die "no cache host: pass --host user@host or set HITECH_CACHE_HOST (or use --no-push)"
  fi

  log "repository: $repo"
  log "targets:    ${targets[*]}"
  [ -n "$host" ] && log "cache host: $host"
  [ "$no_push" -eq 1 ] && log "mode:       build only (--no-push)"
  [ "$no_sign" -eq 1 ] && log "mode:       signing disabled (--no-sign)"

  cd "$repo"

  local name attr out
  for name in "${targets[@]}"; do
    attr=$(target_attr "$name")
    log "building $name ($attr)"

    if [ "$dry_run" -eq 1 ]; then
      printf '  nix build %s --no-link --print-out-paths\n' "$attr"
      continue
    fi

    out=$("${NIX[@]}" build "$attr" --no-link --print-out-paths -L)
    [ -n "$out" ] || die "build produced no output for $name"
    log "built $out"

    if [ "$no_sign" -eq 0 ] && [ -n "$sign_key" ]; then
      sign_closure "$sign_key" "$out"
    fi

    if [ "$no_push" -eq 0 ]; then
      push_closure "$host" "$out"
      if [ "$no_sign" -eq 0 ] && [ -z "$sign_key" ]; then
        sign_closure_remote "$host" "$remote_key" "$out"
      fi
      verify_remote "$host" "$out"
    fi
  done

  log "done"
}

main "$@"

#!/usr/bin/env bash
# Verify that a HiTech-OS ISO carries the expected hitech-installer payload.
#
# Usage:
#   ./scripts/e2e/check-iso-contents.sh NETINSTALLER.iso LIVE.iso [INSTALLER_STORE_BASENAME]
#
# The optional third argument is the expected store basename, for example:
#   i5g7srfdxxfgy3qmlwmmjigs4rb2b4sp-hitech-installer-2.0.0
#
# Mounting requires root; the script re-executes mount/umount through sudo when
# it is not already root.
set -euo pipefail

usage() {
  sed -n '2,12p' "$0" | sed 's/^# \{0,1\}//'
}

if [[ $# -lt 2 || $# -gt 3 ]]; then
  usage >&2
  exit 2
fi

net_iso=$1
live_iso=$2
expected=${3:-}

for iso in "$net_iso" "$live_iso"; do
  [[ -f "$iso" ]] || { echo "missing ISO: $iso" >&2; exit 1; }
done

if [[ $(id -u) -eq 0 ]]; then
  as_root() { "$@"; }
else
  as_root() { sudo -- "$@"; }
fi

work=$(mktemp -d)
iso_mnt="$work/iso"
store_mnt="$work/store"
mkdir -p "$iso_mnt" "$store_mnt"
iso_mounted=0
store_mounted=0

cleanup() {
  if [[ $store_mounted -eq 1 ]]; then as_root umount "$store_mnt" || true; fi
  if [[ $iso_mounted -eq 1 ]]; then as_root umount "$iso_mnt" || true; fi
  rm -rf "$work"
}
trap cleanup EXIT

check_iso() {
  local label=$1 iso=$2 squash
  echo "== $label: $iso"
  as_root mount -o loop,ro "$iso" "$iso_mnt"
  iso_mounted=1

  squash=$(find "$iso_mnt" -maxdepth 1 -name 'nix-store.squashfs' -print -quit)
  if [[ -z "$squash" ]]; then
    echo "  no nix-store.squashfs found" >&2
    return 1
  fi

  as_root mount -o loop,ro "$squash" "$store_mnt"
  store_mounted=1

  local found
  found=$(find "$store_mnt" -maxdepth 1 -type d -name '*hitech-installer*' -printf '%f\n' | sort)
  if [[ -z "$found" ]]; then
    echo "  no hitech-installer store path found" >&2
    return 1
  fi
  echo "$found" | sed 's/^/  /'

  if [[ -n "$expected" ]] && ! grep -Fxq "$expected" <<<"$found"; then
    echo "  expected payload not found: $expected" >&2
    return 1
  fi

  as_root umount "$store_mnt"
  store_mounted=0
  as_root umount "$iso_mnt"
  iso_mounted=0
}

check_iso NetInstaller "$net_iso"
check_iso Live "$live_iso"

echo "ISO payload check: PASS"

  if cmd_exists openssl; then
    openssl rand -hex "$bytes"
  else
    od -An -N "$bytes" -tx1 /dev/urandom | tr -d ' \n'
  fi
}

ensure_root() {
  if [[ "$ROOT" == "/" && $(id -u) -ne 0 ]]; then
    die "run as root (sudo) or use --dry-run/--verify"
  fi
}

write_if_missing() {
  local path=$1 mode=$2 content=$3 target
  target=$(root_path "$path")
  if [[ -f "$target" ]]; then
    return 0
  fi
  write_file "$path" "$mode" "$content"
}

write_file() {
  local path mode content target tmp
  path=$1; mode=$2; content=$3
  target=$(root_path "$path")
  if [[ $DRY_RUN -eq 1 ]]; then
    log "[dry-run] write $target ($mode, ${#content} bytes)"
    return 0
  fi
  mkdir -p -- "$(dirname -- "$target")"
  if [[ -f "$target" ]] && cmp -s <(printf '%b' "$content") "$target"; then
    return 0
  fi
  tmp=$(mktemp "${target}.tmp.XXXXXX")
  printf '%b' "$content" >"$tmp"
  chmod "$mode" "$tmp"
  mv -f -- "$tmp" "$target"
}

append_once() {
  local path marker line target
  path=$1; marker=$2; line=$3
  target=$(root_path "$path")
  if [[ $DRY_RUN -eq 1 ]]; then log "[dry-run] append '$line' to $target"; return 0; fi
  mkdir -p -- "$(dirname -- "$target")"
  touch -- "$target"
  grep -Fq -- "$marker" "$target" 2>/dev/null && return 0
  printf '\n%s\n' "$line" >>"$target"
}

detect_os() {
  local release_file="${HITECH_OS_RELEASE:-/etc/os-release}" id like
  if [[ -n "${WSL_DISTRO_NAME:-}" ]] || [[ "$(uname -r 2>/dev/null || true)" == *microsoft* ]]; then
    WSL=1
  fi
  if [[ -r "$release_file" ]]; then
    # shellcheck disable=SC1090
    id=$(unset ID; . "$release_file" >/dev/null 2>&1; printf '%s' "${ID:-unknown}")
    # shellcheck disable=SC1090
    like=$(unset ID_LIKE; . "$release_file" >/dev/null 2>&1; printf '%s' "${ID_LIKE:-}")
    # shellcheck disable=SC1090
    OS_NAME=$(unset NAME; . "$release_file" >/dev/null 2>&1; printf '%s' "${NAME:-unknown}")
    # shellcheck disable=SC1090
    OS_VERSION=$(unset VERSION_ID; . "$release_file" >/dev/null 2>&1; printf '%s' "${VERSION_ID:-unknown}")
    OS_ID=$id
    OS_LIKE=$like
  fi
  case " $OS_ID $OS_LIKE " in
    *" nixos "*) DISTRO_FAMILY="nixos" ;;
    *" debian "*|*" ubuntu "*) DISTRO_FAMILY="debian" ;;
    *" arch "*) DISTRO_FAMILY="arch" ;;
    *" fedora "*) DISTRO_FAMILY="fedora" ;;
    *)
      case "$OS_ID" in
        nixos) DISTRO_FAMILY="nixos" ;;
        debian|ubuntu|linuxmint|pop) DISTRO_FAMILY="debian" ;;
        arch|manjaro|endeavouros) DISTRO_FAMILY="arch" ;;
        fedora|rhel|centos) DISTRO_FAMILY="fedora" ;;
        *) DISTRO_FAMILY="unknown" ;;
      esac
      ;;
  esac
  if [[ $WSL -eq 1 ]]; then OS_ID="wsl2"; fi
  SYSTEMD=0
  [[ -d /run/systemd/system ]] && SYSTEMD=1
}

detect_hardware() {
  local mem_kb gpu_line vram_file vram_bytes cpuinfo
  CPU_ARCH=$(uname -m 2>/dev/null || printf 'unknown')
  cpuinfo="${HITECH_CPUINFO:-/proc/cpuinfo}"
  if [[ -r "$cpuinfo" ]]; then
    CPU_VENDOR=$(awk -F: '/^vendor_id/ {gsub(/^[ \t]+|[ \t]+$/, "", $2); print $2; exit}' "$cpuinfo")
    CPU_MODEL=$(awk -F: '/^model name/ {gsub(/^[ \t]+|[ \t]+$/, "", $2); print $2; exit}' "$cpuinfo")
    if [[ -z "$CPU_VENDOR" ]]; then
      case "$CPU_ARCH" in
        aarch64|arm64|armv7l) CPU_VENDOR="ARM" ;;
        *) CPU_VENDOR="unknown" ;;
      esac
    fi
  fi
  CPU_VENDOR=${CPU_VENDOR:-unknown}
  CPU_MODEL=${CPU_MODEL:-unknown}
  mem_kb=$(awk '/^MemTotal:/ {print $2; exit}' /proc/meminfo 2>/dev/null || printf '0')
  RAM_MIB=$(( ${mem_kb:-0} / 1024 ))
  if cmd_exists lspci; then
    gpu_line=$(lspci 2>/dev/null | grep -Ei 'VGA compatible controller|3D controller|Display controller' | head -1 || true)
    GPU_MODEL=${gpu_line#*: }
    case "$gpu_line" in
      *NVIDIA*|*nvidia*) GPU_VENDOR="nvidia" ;;
      *AMD*|*Advanced*Micro*Devices*|*ATI*) GPU_VENDOR="amd" ;;
      *Intel*) GPU_VENDOR="intel" ;;
      *) GPU_VENDOR="unknown" ;;
    esac
  fi
  GPU_MODEL=${GPU_MODEL:-unknown}
  for vram_file in /sys/class/drm/card*/device/mem_info_vram_total; do
    [[ -r "$vram_file" ]] || continue
    vram_bytes=$(cat "$vram_file" 2>/dev/null || printf '0')
    if (( ${vram_bytes:-0} / 1024 / 1024 > VRAM_MIB )); then
      VRAM_MIB=$(( vram_bytes / 1024 / 1024 ))
    fi
  done
  if cmd_exists nvidia-smi; then
    local nv
    nv=$(nvidia-smi --query-gpu=memory.total --format=csv,noheader,nounits 2>/dev/null | head -1 | tr -dc '0-9' || true)
    if [[ -n "$nv" ]] && (( nv > VRAM_MIB )); then VRAM_MIB=$nv; GPU_VENDOR="nvidia"; fi
  fi
}

print_detection() {
  printf '%sHiTech-OS bootstrap detection%s\n' "$C_BOLD" "$C_RESET"
  printf '  OS:       %s %s (%s)\n' "$OS_NAME" "$OS_VERSION" "$OS_ID"
  printf '  Family:   %s%s\n' "$DISTRO_FAMILY" "$([[ $WSL -eq 1 ]] && printf ' / WSL2')"
  printf '  Systemd:  %s\n' "$([[ $SYSTEMD -eq 1 ]] && printf yes || printf no)"
  printf '  CPU:      %s (%s)\n' "$CPU_MODEL" "$CPU_ARCH"
  printf '  CPU ven:  %s\n' "$CPU_VENDOR"
  printf '  RAM:      %s MiB\n' "$RAM_MIB"
  printf '  GPU:      %s (%s)\n' "$GPU_MODEL" "$GPU_VENDOR"
  printf '  VRAM:     %s MiB\n' "$VRAM_MIB"
  printf '  Profile:  %s\n' "$PROFILE"
}

resolve_source() {
  if [[ -n "$SOURCE_DIR" ]]; then
    [[ -f "$SOURCE_DIR/flake.nix" ]] || die "--source does not contain flake.nix: $SOURCE_DIR"
    SOURCE_DIR=$(cd -- "$SOURCE_DIR" && pwd)
    FLAKE_REF="path:$SOURCE_DIR"
    return
  fi
  if [[ -f "$SCRIPT_DIR/../flake.nix" ]]; then
    SOURCE_DIR=$(cd -- "$SCRIPT_DIR/.." && pwd)
    FLAKE_REF="path:$SOURCE_DIR"
    return
  fi
  if [[ -n "${HITECH_FLAKE:-}" ]]; then
    FLAKE_REF="$HITECH_FLAKE"
    return
  fi
  SOURCE_DIR="/opt/hitech-os"
  FLAKE_REF="github:HiTechTN/HiTech-OS"
  if [[ $DRY_RUN -eq 1 ]]; then return; fi
  if [[ ! -d "$SOURCE_DIR/.git" ]]; then
    run_root mkdir -p "$(dirname "$SOURCE_DIR")"
    run_root git clone --depth 1 https://github.com/HiTechTN/HiTech-OS.git "$SOURCE_DIR"
  else
    run_root git -C "$SOURCE_DIR" fetch --depth 1 origin main
    run_root git -C "$SOURCE_DIR" reset --hard FETCH_HEAD
  fi
  FLAKE_REF="path:$SOURCE_DIR"
}

install_nix() {
  local installer
  if cmd_exists nix; then
    ok "Nix already installed: $(nix --version 2>/dev/null || true)"
  elif [[ $DRY_RUN -eq 1 ]]; then
    log "[dry-run] install multi-user Nix from https://nixos.org/nix/install"
  else
    cmd_exists curl || die "curl is required to install Nix"
    installer=$(mktemp)
    log "downloading the official Nix installer"
    curl -fsSL https://nixos.org/nix/install -o "$installer"
    if [[ $WSL -eq 1 && $SYSTEMD -eq 0 ]]; then
      warn "WSL2 without systemd: using single-user Nix (enable systemd=true in /etc/wsl.conf for daemon mode)"
      run_root sh "$installer" --no-daemon
    elif [[ $ASSUME_YES -eq 1 ]]; then
      NIX_INSTALLER_YES=1 run_root sh "$installer" --daemon
    else
      run_root sh "$installer" --daemon
    fi
    rm -f "$installer"
    # Make the daemon's nix binary visible to the rest of this process.
    export PATH="/nix/var/nix/profiles/default/bin:$PATH"
    command -v nix >/dev/null 2>&1 || die "Nix installer finished but nix is not on PATH"
  fi
  if [[ $DRY_RUN -eq 0 ]]; then
    if [[ ! -f "$(root_path /etc/nix/nix.conf)" ]]; then
      write_file /etc/nix/nix.conf 0644 "experimental-features = nix-command flakes
trusted-users = root ${SUDO_USER:-}
sandbox = relaxed
"
    else
      append_once /etc/nix/nix.conf "experimental-features" "experimental-features = nix-command flakes"
      append_once /etc/nix/nix.conf "trusted-users" "trusted-users = root ${SUDO_USER:-}"
      append_once /etc/nix/nix.conf "sandbox" "sandbox = relaxed"
    fi
    if [[ $SYSTEMD -eq 1 ]]; then run_root systemctl restart nix-daemon.service >/dev/null 2>&1 || true; fi
  else
    log "[dry-run] enable nix-command + flakes in /etc/nix/nix.conf"
  fi
}

ensure_rsync() {
  if cmd_exists rsync; then return; fi
  if [[ $DRY_RUN -eq 1 ]]; then log "[dry-run] install rsync"; return; fi
  case "$DISTRO_FAMILY" in
    debian) run_root apt-get update; run_root apt-get install -y rsync ;;
    arch) run_root pacman -Sy --noconfirm rsync ;;
    fedora) run_root dnf install -y rsync ;;
    *) die "rsync is required; install it and rerun" ;;
  esac
}

backup_nixos() {
  local stamp backup
  stamp=$(date -u +%Y%m%dT%H%M%SZ)
  backup="$(root_path "$BACKUP_ROOT")/nixos-$stamp"
  run_root mkdir -p "$backup"
  if [[ -d "$(root_path /etc/nixos)" ]]; then
    run_root rsync -a --delete "$(root_path /etc/nixos)/" "$backup/"
    ok "backed up /etc/nixos to $backup"
  else
    warn "/etc/nixos does not exist; no host configuration to back up"
  fi
}

morph_nixos() {
  local target host_src
  target="$(root_path /etc/nixos/hitech-os)"
  host_src="$(root_path /etc/nixos)"
  ensure_rsync
  backup_nixos
  run_root mkdir -p "$target" "$target/host"
  if [[ "$(realpath -m -- "$SOURCE_DIR")" != "$(realpath -m -- "$target")" ]]; then
    run_root rsync -aL --delete --exclude .git --exclude host "$SOURCE_DIR/" "$target/"
  fi
  # Preserve the host's generated hardware/bootloader/network modules. They
  # are evaluated by os/profiles/* through flake's hostConfiguration.
  if [[ -d "$host_src" ]]; then
    run_root rsync -a --delete --exclude hitech-os --exclude host "$host_src/" "$target/host/"
  fi
  [[ $DRY_RUN -eq 1 ]] || [[ -f "$target/flake.nix" ]] || die "source copy did not produce $target/flake.nix"
  log "running nixos-rebuild boot --flake $target#$PROFILE"
  run_root nixos-rebuild boot --flake "$target#$PROFILE"
  ok "HiTech-OS $PROFILE staged for the next boot"
}

profile_install_one() {
  local pkg=$1 flake=$2 nixbin=$3
  if [[ $DRY_RUN -eq 1 ]]; then
    log "[dry-run] nix profile install $flake#$pkg"
    return 0
  fi
  run_root "$nixbin" profile remove --profile "$NIX_PROFILE" "$pkg" >/dev/null 2>&1 || true
  run_root "$nixbin" profile install --profile "$NIX_PROFILE" "$flake#$pkg"
}

install_nix_packages() {
  local nixbin pkg
  nixbin=$(command -v nix) || die "nix is not installed"
  for pkg in $(profile_packages); do
    profile_install_one "$pkg" "$FLAKE_REF" "$nixbin"
  done
}

install_services() {
  local bin profiles
  [[ $NO_SERVICES -eq 1 ]] && { warn "systemd units skipped (--no-services)"; return; }
  [[ $SYSTEMD -eq 1 ]] || { warn "systemd not available; service units were not enabled"; return; }
  profiles="/nix/var/nix/profiles/default/bin"
  for user in hitech-store genio mosquitto influxdb2 telegraf; do
    if ! getent group "$user" >/dev/null 2>&1; then
      run_root groupadd --system "$user" || true
    fi
    if ! id "$user" >/dev/null 2>&1; then
      run_root useradd --system --gid "$user" --home "/var/lib/$user" --shell /usr/sbin/nologin "$user" || true
    fi
  done
  run_root mkdir -p /var/lib/hitech-store /var/lib/genio /var/lib/mosquitto /var/lib/influxdb2 /var/lib/telegraf /etc/hitechos
  write_if_missing /etc/hitechos/hitech-store.env 0600 "SECRET_KEY=$(random_hex 32)\nSTORAGE_DIR=/var/lib/hitech-store/storage\nAGENT_SHARED_SECRET=$(random_hex 24)\nDEV_OTP_FALLBACK=0\nCOOKIE_SECURE=0\n"
  write_file /etc/hitechos/genio.env 0640 "HITECH_STORE_URL=http://127.0.0.1:8095\n"
  write_file /etc/hitechos/mosquitto.conf 0644 "listener 1883 127.0.0.1\nallow_anonymous true\npersistence true\npersistence_location /var/lib/mosquitto/\n"
  write_file /etc/hitechos/influxdb2.conf 0644 "[http]\n  bind-address = \"127.0.0.1:8086\"\n\n[storage]\n  bolt-path = \"/var/lib/influxdb2/influxd.bolt\"\n  engine-path = \"/var/lib/influxdb2/engine\"\n"
  write_file /etc/hitechos/telegraf.conf 0644 "[[inputs.mqtt_consumer]]\n  servers = [\"tcp://127.0.0.1:1883\"]\n  topics = [\"hitechos/+/+/+\"]\n  data_format = \"json\"\n[[outputs.influxdb_v2]]\n  urls = [\"http://127.0.0.1:8086\"]\n  token = \"\${INFLUX_TOKEN}\"\n  organization = \"HiTechTN\"\n  bucket = \"hitechos\"\n"
  write_file /etc/systemd/system/hitech-store.service 0644 "[Unit]\nDescription=HiTech Store (HiTech Drive)\nAfter=network.target\n[Service]\nUser=hitech-store\nGroup=hitech-store\nWorkingDirectory=/var/lib/hitech-store\nEnvironmentFile=/etc/hitechos/hitech-store.env\nEnvironment=LOG_DIR=/var/lib/hitech-store/logs\nExecStart=$profiles/hitech-store app:app --host 127.0.0.1 --port 8095 --workers 2\nRestart=on-failure\nNoNewPrivileges=true\nProtectSystem=full\n[Install]\nWantedBy=multi-user.target\n"
  write_file /etc/systemd/system/genio.service 0644 "[Unit]\nDescription=Genio local runtime and dashboard\nAfter=network.target hitech-store.service\n[Service]\nUser=genio\nGroup=genio\nEnvironmentFile=/etc/hitechos/genio.env\nEnvironment=GENIO_HOST=127.0.0.1\nEnvironment=GENIO_PORT=8765\nEnvironment=GENIO_STATE_DIR=/var/lib/genio\nExecStart=$profiles/genio-daemon\nRestart=on-failure\nNoNewPrivileges=true\n[Install]\nWantedBy=multi-user.target\n"
  write_file /etc/systemd/system/hitechos-ai-daemon.service 0644 "[Unit]\nDescription=HiTech-OS local inference daemon\nAfter=network.target\n[Service]\nType=oneshot\nRemainAfterExit=yes\nExecStart=$profiles/hitechos-ai-daemon\n[Install]\nWantedBy=multi-user.target\n"
  write_file /etc/systemd/system/mosquitto.service 0644 "[Unit]\nDescription=Mosquitto MQTT broker (HiTech-OS)\nAfter=network.target\n[Service]\nUser=mosquitto\nGroup=mosquitto\nExecStart=$profiles/mosquitto -c /etc/hitechos/mosquitto.conf\nRestart=on-failure\n[Install]\nWantedBy=multi-user.target\n"
  write_file /etc/systemd/system/influxdb2.service 0644 "[Unit]\nDescription=InfluxDB 2 (HiTech-OS telemetry)\nAfter=network.target\n[Service]\nUser=influxdb2\nGroup=influxdb2\nEnvironment=INFLUXD_BOLT_PATH=/var/lib/influxdb2/influxd.bolt\nEnvironment=INFLUXD_ENGINE_PATH=/var/lib/influxdb2/engine\nEnvironment=INFLUXD_HTTP_BIND_ADDRESS=127.0.0.1:8086\nExecStart=$profiles/influxd\nRestart=on-failure\n[Install]\nWantedBy=multi-user.target\n"
  write_file /etc/systemd/system/telegraf.service 0644 "[Unit]\nDescription=Telegraf MQTT to InfluxDB bridge (HiTech-OS)\nAfter=network.target mosquitto.service influxdb2.service\n[Service]\nUser=telegraf\nGroup=telegraf\nEnvironmentFile=-/etc/hitechos/telegraf.env\nExecStart=$profiles/telegraf --config /etc/hitechos/telegraf.conf\nRestart=on-failure\n[Install]\nWantedBy=multi-user.target\n"
  if [[ $DRY_RUN -eq 1 ]]; then return; fi
  run_root systemctl daemon-reload
  for unit in $(service_units); do
    run_root systemctl enable --now "$unit.service" || warn "$unit.service failed to start"
  done
}

verify_install() {
  local failures=0 path
  log "verifying HiTech-OS installation"
  detect_os; detect_hardware; print_detection
  for path in /etc/hitechos-release /etc/hitechos/os-release /etc/xdg/foot/foot.ini; do
    if [[ -f "$(root_path "$path")" ]]; then ok "$path"; else warn "missing $path"; failures=$((failures + 1)); fi
  done
  if [[ "$DISTRO_FAMILY" == "nixos" ]]; then
    if [[ -f "$(root_path /etc/nixos/hitech-os/flake.nix)" ]]; then ok "NixOS flake source"; else warn "missing /etc/nixos/hitech-os/flake.nix"; failures=$((failures + 1)); fi
    if cmd_exists nixos-rebuild; then ok "nixos-rebuild available"; else warn "nixos-rebuild missing"; failures=$((failures + 1)); fi
  else
    if [[ -x "$(root_path /nix/var/nix/profiles/default/bin/hitech-shell)" ]]; then ok "hitech-shell installed"; else warn "hitech-shell not in system profile"; failures=$((failures + 1)); fi
    if [[ -f "$(root_path /usr/share/wayland-sessions/hitech-shell.desktop)" ]]; then ok "Wayland session registered"; else warn "Wayland session missing"; failures=$((failures + 1)); fi
  fi
  if [[ $SYSTEMD -eq 1 ]]; then
    for unit in $(service_units); do
      if systemctl is-enabled --quiet "$unit.service" 2>/dev/null; then ok "$unit enabled"; else warn "$unit not enabled"; failures=$((failures + 1)); fi
    done
  fi
  (( failures == 0 )) || die "$failures verification check(s) failed"
  ok "verification passed"
}

parse_args() {
  while [[ $# -gt 0 ]]; do
    case "$1" in
      --profile) [[ $# -ge 2 ]] || die "--profile requires a value"; PROFILE=$2; shift 2 ;;
      --profile=*) PROFILE=${1#*=}; shift ;;
      --source) [[ $# -ge 2 ]] || die "--source requires a value"; SOURCE_DIR=$2; shift 2 ;;
      --source=*) SOURCE_DIR=${1#*=}; shift ;;
      --dry-run) DRY_RUN=1; shift ;;
      --verify) VERIFY_ONLY=1; shift ;;
      --no-services) NO_SERVICES=1; shift ;;
      --root) [[ $# -ge 2 ]] || die "--root requires a value"; ROOT=$2; shift 2 ;;
      --root=*) ROOT=${1#*=}; shift ;;
      -y|--yes) ASSUME_YES=1; shift ;;
      -h|--help) usage; exit 0 ;;
      *) die "unknown argument: $1 (try --help)" ;;
    esac
  done
  valid_profile "$PROFILE" || die "invalid profile: $PROFILE"
  ROOT="${ROOT%/}"; [[ -n "$ROOT" ]] || ROOT="/"
}

main() {
  parse_args "$@"
  if [[ $DRY_RUN -eq 0 && $VERIFY_ONLY -eq 0 && "$ROOT" == "/" ]]; then
    mkdir -p "$(dirname "$LOG_FILE")" 2>/dev/null || true
    exec > >(tee -a "$LOG_FILE") 2>&1
  fi
  detect_os
  detect_hardware
  print_detection
  if [[ $VERIFY_ONLY -eq 1 ]]; then verify_install; return; fi
  log "profile=$PROFILE family=$DISTRO_FAMILY wsl=$WSL systemd=$SYSTEMD dry_run=$DRY_RUN"
  if [[ $DRY_RUN -eq 1 ]]; then
    resolve_source
    log "[dry-run] would install Nix and apply the $PROFILE profile"
    log "[dry-run] source: $FLAKE_REF"
    log "[dry-run] packages: $(profile_packages | tr '\n' ' ')"
    log "[dry-run] desktop: /usr/share/wayland-sessions/hitech-shell.desktop"
    log "[dry-run] services: hitech-store genio hitechos-ai-daemon mosquitto influxdb2 telegraf"
    log "[dry-run] branding: /etc/hitechos-release, /etc/motd, /etc/xdg/foot/foot.ini"
    ok "dry-run complete; no changes made"
    return
  fi
  ensure_root
  resolve_source
  install_nix
  if [[ "$DISTRO_FAMILY" == "nixos" ]]; then
    morph_nixos
  else
    install_nix_packages
    install_desktop
    install_services
  fi
  install_branding
  save_report
  ok "HiTech-OS $PROFILE bootstrap complete"
}

  main "$@"

# compiler; the CUDA backend stays a feature flag (see docs/ROADMAP.md Phase 4).
{
  lib,
  rustPlatform,
}:

rustPlatform.buildRustPackage {
  pname = "hitechos-ai-daemon";
  version = "0.1.0";

  src = lib.cleanSourceWith {
    src = ./.;
    filter =
      path: _type:
      let
        base = baseNameOf (toString path);
      in
      base != "target" && base != "result" && base != ".git";
  };

  cargoLock.lockFile = ./Cargo.lock;

  # Stub-backend tests + clippy-clean code are enforced by the CI rust job;
  # running them here too keeps `nix flake check` honest.
  doCheck = true;

  meta = with lib; {
    description = "HiTech-OS AI daemon — pluggable GGUF inference backend (stub by default)";
    homepage = "https://github.com/HiTechTN/HiTech-OS";
    license = licenses.agpl3Plus;
    mainProgram = "hitechos-ai-daemon";
    platforms = platforms.linux;
  };
}
