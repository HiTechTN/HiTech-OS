/// Errors surfaced by the compositor bootstrap or a fatal backend failure.
#[derive(Debug)]
pub enum CompositorError {
    Session(String),
    Display(String),
    Loop(String),
    Backend(String),
}

impl fmt::Display for CompositorError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Session(e) => write!(f, "session backend error: {e}"),
            Self::Display(e) => write!(f, "wayland display error: {e}"),
            Self::Loop(e) => write!(f, "event loop error: {e}"),
            Self::Backend(e) => write!(f, "graphics backend error: {e}"),
        }
    }
}

impl Error for CompositorError {}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct UdevOutputId {

    crtc: crtc::Handle,
}

struct Surface {
    output: Output,
    drm_output: GbmDrmOutput,
}

    renderer: GlesRenderer,
    scanner: DrmScanner,

/// Global compositor state, owned by the calloop event loop.
pub struct HitechShell {
    start_time: Instant,
    socket_name: std::ffi::OsString,
    display_handle: DisplayHandle,
    loop_signal: LoopSignal,
    loop_handle: LoopHandle<'static, HitechShell>,

    space: Space<Window>,
    compositor_state: CompositorState,
    xdg_shell_state: XdgShellState,
    shm_state: ShmState,
    output_manager_state: smithay::wayland::output::OutputManagerState,
    seat_state: SeatState<HitechShell>,
    data_device_state: DataDeviceState,
    popups: PopupManager,
    seat: Seat<HitechShell>,

    session: LibSeatSession,
    devices: HashMap<DrmNode, Device>,

    config: Config,
    paused: bool,
    redraw_queued: bool,
}

#[derive(Default)]
struct ClientState {
    compositor_state: CompositorClientState,
}

impl ClientData for ClientState {
    fn initialized(&self, _client_id: ClientId) {}
    fn disconnected(&self, _client_id: ClientId, _reason: DisconnectReason) {}
}

impl HitechShell {
    fn new(
        config: Config,
        display: Display<HitechShell>,
        event_loop: &mut EventLoop<'static, HitechShell>,
        session: LibSeatSession,
        notifier: LibSeatSessionNotifier,
    ) -> Result<Self, CompositorError> {
        let display_handle = display.handle();
        let start_time = Instant::now();

        let compositor_state = CompositorState::new::<Self>(&display_handle);
        let xdg_shell_state = XdgShellState::new::<Self>(&display_handle);
        let shm_state = ShmState::new::<Self>(&display_handle, vec![]);
        let output_manager_state =
            smithay::wayland::output::OutputManagerState::new_with_xdg_output::<Self>(
                &display_handle,
            );
        let mut seat_state = SeatState::new();
        let data_device_state = DataDeviceState::new::<Self>(&display_handle);
        let popups = PopupManager::default();

        let mut seat: Seat<Self> = seat_state.new_wl_seat(&display_handle, "seat0");
        seat.add_keyboard(Default::default(), 200, 25)
            .map_err(|e| CompositorError::Backend(format!("cannot add keyboard: {e}")))?;
        seat.add_pointer();

        let loop_handle = event_loop.handle();
        let loop_signal = event_loop.get_signal();

        let listening_socket = ListeningSocketSource::new_auto()
            .map_err(|e| CompositorError::Display(format!("cannot create wayland socket: {e}")))?;
        let socket_name = listening_socket.socket_name().to_os_string();

        let socket_handle = loop_handle.clone();
        let _ = socket_handle.insert_source(
            listening_socket,
            move |client_stream, _, state: &mut HitechShell| {
                if let Err(e) = state
                    .display_handle
                    .insert_client(client_stream, Arc::new(ClientState::default()))
                {
                    warn!("cannot insert wayland client: {e}");
                }
            },
        );

        let _ = loop_handle.insert_source(
            Generic::new(display, Interest::READ, Mode::Level),
            |_, display, state: &mut HitechShell| {
                // SAFETY: the display lives in this event source until the loop exits.
                let display = unsafe { display.get_mut() };
                if let Err(e) = display.dispatch_clients(state) {
                    error!("wayland dispatch error: {e}");
                }
                // Registry replies and configure events are queued on the
                // display, not sent by dispatch_clients alone. Without this
                // flush a newly connected client blocks forever in
                // wl_display.sync before it can create a surface.
                if let Err(e) = display.flush_clients() {
                    error!("wayland flush error: {e}");
                }
                Ok(PostAction::Continue)
            },
        );

        std::env::set_var("WAYLAND_DISPLAY", &socket_name);
        info!(
            "hitech-shell compositor {} on {}",
            env!("CARGO_PKG_VERSION"),
            socket_name.to_string_lossy()
        );

        let mut state = Self {
            start_time,
            socket_name,
            display_handle,
            loop_signal,
            loop_handle,
            space: Space::default(),
            compositor_state,
            xdg_shell_state,
            shm_state,
            output_manager_state,
            seat_state,
            data_device_state,
            popups,
            seat,
            session,
            devices: HashMap::new(),
            config,
            paused: false,
            redraw_queued: false,
        };

        let _ = state
            .loop_handle
            .insert_source(notifier, |event, _, state: &mut HitechShell| match event {
                SessionEvent::PauseSession => {
                    info!("session paused; suspending rendering");
                    state.paused = true;
                }
                SessionEvent::ActivateSession => {
                    info!("session activated; resuming rendering");
                    state.paused = false;
                    state.queue_redraw();
                }
            });

        state.init_backends();
        Ok(state)
    }

    fn init_backends(&mut self) {
        let seat_name = self.session.seat();

        let _ =
            self.loop_handle.insert_source(
                LibinputInputBackend::new({
                    let mut context = Libinput::new_with_udev::<
                        LibinputSessionInterface<LibSeatSession>,
                    >(self.session.clone().into());
                    if context.udev_assign_seat(&seat_name).is_err() {
                        error!("cannot assign libinput seat {seat_name}");
                    }
                    context
                }),
                |mut event, _, state: &mut HitechShell| {
                    if let InputEvent::DeviceAdded { device } = &mut event {
                        if device.has_capability(DeviceCapability::Keyboard) {
                            if let Some(keyboard) = state.seat.get_keyboard() {
                                device.led_update(keyboard.led_state().into());
                            }
                        }
                    }
                    state.process_input_event(event);
                },
            );

        let backend = match UdevBackend::new(&seat_name) {
            Ok(backend) => backend,
            Err(e) => {
                error!("cannot initialize udev backend: {e}");
                return;
            }
        };

        let devices: Vec<(DrmNode, PathBuf)> = backend
            .device_list()
            .filter_map(|(device_id, path)| {
                DrmNode::from_dev_id(device_id)
                    .ok()
                    .map(|node| (node, path.to_owned()))
            })
            .collect();

        let _ = self
            .loop_handle
            .insert_source(backend, |event, _, state: &mut HitechShell| {
                state.on_udev_event(event)
            });

        for (node, path) in devices {
            self.on_udev_added(node, &path);
        }
    }

    fn on_udev_event(&mut self, event: UdevEvent) {
        match event {
            UdevEvent::Added { device_id, path } => {
                if let Ok(node) = DrmNode::from_dev_id(device_id) {
                    self.on_udev_added(node, &path);
                }
            }
            UdevEvent::Changed { device_id } => {
                if let Ok(node) = DrmNode::from_dev_id(device_id) {
                    self.device_changed(node);
                }
            }
            UdevEvent::Removed { device_id } => {
                if let Ok(node) = DrmNode::from_dev_id(device_id) {
                    self.device_removed(node);
                }
            }
        }
    }

    fn on_udev_added(&mut self, node: DrmNode, path: &Path) {
        if node.ty() != NodeType::Primary || self.devices.contains_key(&node) {
            return;
        }

        let fd: OwnedFd = match self.session.open(
            path,
            OFlags::RDWR | OFlags::CLOEXEC | OFlags::NOCTTY | OFlags::NONBLOCK,
        ) {
            Ok(fd) => fd,
            Err(e) => {
                warn!("cannot open {} via libseat: {e}", path.display());
                return;
            }
        };
        let fd = DrmDeviceFd::new(DeviceFd::from(fd));

        let (drm, notifier) = match DrmDevice::new(fd.clone(), true) {
            Ok(pair) => pair,
            Err(e) => {
                warn!("cannot initialize DRM device {}: {e}", path.display());
                return;
            }
        };

        let notifier_token = match self.loop_handle.insert_source(
            notifier,
            move |event, _, state: &mut HitechShell| match event {
                DrmEvent::VBlank(crtc) => state.frame_finish(node, crtc),
                DrmEvent::Error(e) => error!("DRM error on {node}: {e}"),
            },
        ) {
            Ok(token) => token,
            Err(e) => {
                warn!("cannot register DRM notifier: {e}");
                return;
            }
        };

        let gbm = match GbmDevice::new(fd) {
            Ok(gbm) => gbm,
            Err(e) => {
                warn!("cannot initialize GBM device: {e}");
                return;
            }
        };

        let renderer = {
            let egl_display = match unsafe { EGLDisplay::new(gbm.clone()) } {
                Ok(display) => display,
                Err(e) => {
                    warn!("cannot create EGL display: {e}");
                    return;
                }
            };
            let context = match EGLContext::new(&egl_display) {
                Ok(context) => context,
                Err(e) => {
                    warn!("cannot create EGL context: {e}");
                    return;
                }
            };
            match unsafe { GlesRenderer::new(context) } {
                Ok(renderer) => renderer,
                Err(e) => {
                    warn!("cannot create GLES renderer: {e}");
                    return;
                }
            }
        };

        let allocator = GbmAllocator::new(
            gbm.clone(),
            GbmBufferFlags::RENDERING | GbmBufferFlags::SCANOUT,
        );
        let exporter = GbmFramebufferExporter::new(gbm.clone(), Some(node));
        let render_formats: Vec<_> = renderer.dmabuf_formats().iter().copied().collect();
        let manager = DrmOutputManager::new(
            drm,
            allocator,
            exporter,
            Some(gbm),
            SUPPORTED_FORMATS.iter().copied(),
            render_formats,
        );

        info!("DRM device {node} ready ({})", path.display());
        self.devices.insert(
            node,
            Device {
                manager,
                renderer,
                scanner: DrmScanner::new(),
                surfaces: HashMap::new(),
                notifier_token,
            },
        );

        self.device_changed(node);
    }

    fn device_changed(&mut self, node: DrmNode) {
        let events = {
            let Some(device) = self.devices.get_mut(&node) else {
                return;
            };
            match device.scanner.scan_connectors(device.manager.device()) {
                Ok(events) => events.into_iter().collect::<Vec<_>>(),
                Err(e) => {
                    warn!("cannot scan connectors on {node}: {e}");
                    return;
                }
            }
        };

        for event in events {
            match event {
                DrmScanEvent::Connected {
                    connector,
                    crtc: Some(crtc),
                } => self.connector_connected(node, connector, crtc),
                DrmScanEvent::Disconnected {
                    connector,
                    crtc: Some(crtc),
                } => self.connector_disconnected(node, &connector, crtc),
                _ => {}
            }
        }
    }

    fn connector_connected(
        &mut self,
        node: DrmNode,
        connector: connector::Info,
        crtc: crtc::Handle,
    ) {
        let Some(device) = self.devices.get_mut(&node) else {
            return;
        };

        let Some(mode) = connector
            .modes()
            .iter()
            .find(|mode| mode.mode_type().contains(ModeTypeFlags::PREFERRED))
            .or_else(|| connector.modes().first())
            .copied()
        else {
            warn!("connector {:?} has no modes", connector.handle());
            return;
        };

        let output_name = format!(
            "{}-{}",
            connector.interface().as_str(),
            connector.interface_id()
        );
        let (phys_w, phys_h) = connector.size().unwrap_or((0, 0));
        let output = Output::new(
            output_name.clone(),
            PhysicalProperties {
                size: (phys_w as i32, phys_h as i32).into(),
                subpixel: connector.subpixel().into(),
                make: "HiTech".to_string(),
                model: "Panel".to_string(),
            },
        );
        let _global = output.create_global::<HitechShell>(&self.display_handle);
        let wl_mode = WlMode::from(mode);

        let x = self.space.outputs().fold(0, |acc, o| {
            acc + self.space.output_geometry(o).map_or(0, |g| g.size.w)
        });
        let position: Point<i32, Logical> = (x, 0).into();

        output.set_preferred(wl_mode);
        output.change_current_state(Some(wl_mode), None, None, Some(position));
        self.space.map_output(&output, position);
        output.user_data().insert_if_missing(|| UdevOutputId {
            device_id: node,
            crtc,
        });

        let planes = device.manager.device().planes(&crtc).ok();
        let render_elements =
            DrmOutputRenderElements::<GlesRenderer, ShellRenderElement>::default();

        let Device {
            manager, renderer, ..
        } = device;
        let drm_output = match manager.initialize_output::<_, ShellRenderElement>(
            crtc,
            mode,
            &[connector.handle()],
            &output,
            planes,
            renderer,
            &render_elements,
        ) {
            Ok(drm_output) => drm_output,
            Err(e) => {
                warn!("cannot initialize output {output_name}: {e}");
                self.space.unmap_output(&output);
                return;
            }
        };

        device.surfaces.insert(crtc, Surface { output, drm_output });
        info!("output {output_name} enabled on {node}/{crtc:?}");
        self.queue_redraw();
    }

    fn connector_disconnected(
        &mut self,
        node: DrmNode,
        connector: &connector::Info,
        crtc: crtc::Handle,
    ) {
        let Some(device) = self.devices.get_mut(&node) else {
            return;
        };
        if let Some(surface) = device.surfaces.remove(&crtc) {
            self.space.unmap_output(&surface.output);
            info!(
                "output {} disconnected ({:?})",
                surface.output.name(),
                connector.handle()
            );
            self.queue_redraw();
        }
    }

    fn device_removed(&mut self, node: DrmNode) {
        if let Some(device) = self.devices.remove(&node) {
            self.loop_handle.remove(device.notifier_token);
            let outputs: Vec<Output> = device
                .surfaces
                .values()
                .map(|surface| surface.output.clone())
                .collect();
            for output in outputs {
                self.space.unmap_output(&output);
            }
            info!("DRM device {node} removed");
        }
    }

    fn frame_finish(&mut self, node: DrmNode, crtc: crtc::Handle) {
        if let Some(device) = self.devices.get_mut(&node) {
            if let Some(surface) = device.surfaces.get_mut(&crtc) {
                if let Err(e) = surface.drm_output.frame_submitted() {
                    warn!("frame_submitted failed on {node}/{crtc:?}: {e}");
                }
            }
        }
    }

    fn queue_redraw(&mut self) {
        if self.redraw_queued {
            return;
        }
        self.redraw_queued = true;
        let _ = self.loop_handle.insert_idle(|state: &mut HitechShell| {
            state.redraw_queued = false;
            state.render_all();
        });
    }

    fn render_all(&mut self) {
        if self.paused {
            return;
        }
        let nodes: Vec<DrmNode> = self.devices.keys().copied().collect();
        for node in nodes {
            let crtcs: Vec<crtc::Handle> = self
                .devices
                .get(&node)
                .map(|device| device.surfaces.keys().copied().collect())
                .unwrap_or_default();
            for crtc in crtcs {
                self.render_output(node, crtc);
            }
        }
    }

    fn render_output(&mut self, node: DrmNode, crtc: crtc::Handle) {
        let Self {
            space,
            devices,
            popups,
            display_handle,
            start_time,
            paused,
            ..
        } = self;
        if *paused {
            return;
        }

        let Some(device) = devices.get_mut(&node) else {
            return;
        };
        let Some(surface) = device.surfaces.get_mut(&crtc) else {
            return;
        };
        let output = surface.output.clone();

        let elements = match space.render_elements_for_output(&mut device.renderer, &output, 1.0) {
            Ok(elements) => elements,
            Err(e) => {
                warn!("cannot build render elements for {}: {e}", output.name());
                return;
            }
        };

        let frame = match surface.drm_output.render_frame(
            &mut device.renderer,
            &elements,
            CLEAR_COLOR,
            FrameFlags::DEFAULT,
        ) {
            Ok(frame) => frame,
            Err(e) => {
                warn!("render failed on {}: {e:?}", output.name());
                return;
            }
        };

        if !frame.is_empty {
            if let Err(e) = surface.drm_output.queue_frame(()) {
                warn!("queue_frame failed on {}: {e}", output.name());
                return;
            }
            let now = start_time.elapsed();
            space.elements().for_each(|window| {
                window.send_frame(&output, now, Some(Duration::ZERO), |_, _| {
                    Some(output.clone())
                })
            });
        }

        space.refresh();
        popups.cleanup();
        let _ = display_handle.flush_clients();
    }

    fn surface_under(&self, pos: Point<f64, Logical>) -> Option<(WlSurface, Point<f64, Logical>)> {
        self.space
            .element_under(pos)
            .and_then(|(window, location)| {
                window
                    .surface_under(pos - location.to_f64(), WindowSurfaceType::ALL)
                    .map(|(surface, point)| (surface, (point + location).to_f64()))
            })
    }

    fn unconstrain_popup(&self, popup: &PopupSurface) {
        let Ok(root) = find_popup_root_surface(&PopupKind::Xdg(popup.clone())) else {
            return;
        };
        let Some(window) = self
            .space
            .elements()
            .find(|window| window.toplevel().map(|t| t.wl_surface()) == Some(&root))
        else {
            return;
        };
        let Some(output) = self.space.outputs().next() else {
            return;
        };
        let Some(output_geo) = self.space.output_geometry(output) else {
            return;
        };
        let Some(window_geo) = self.space.element_geometry(window) else {
            return;
        };

        let mut target = output_geo;
        target.loc -= get_popup_toplevel_coords(&PopupKind::Xdg(popup.clone()));
        target.loc -= window_geo.loc;

        popup.with_pending_state(|state| {
            state.geometry = state.positioner.get_unconstrained_geometry(target);
        });
    }

    fn process_input_event(&mut self, event: InputEvent<LibinputInputBackend>) {
        match event {
            InputEvent::Keyboard { event, .. } => {
                let serial = SERIAL_COUNTER.next_serial();
                let time = Event::time_msec(&event);
                if let Some(keyboard) = self.seat.get_keyboard() {
                    keyboard.input::<(), _>(
                        self,
                        event.key_code(),
                        event.state(),
                        serial,
                        time,
                        |_, _, _| FilterResult::Forward,
                    );
                }
            }
            InputEvent::PointerMotion { event, .. } => {
                let Some(pointer) = self.seat.get_pointer() else {
                    return;
                };
                let mut location = pointer.current_location() + event.delta();
                if let Some(output) = self.space.outputs().next() {
                    if let Some(geo) = self.space.output_geometry(output) {
                        location.x = location.x.clamp(0.0, geo.size.w as f64 - 1.0);
                        location.y = location.y.clamp(0.0, geo.size.h as f64 - 1.0);
                    }
                }
                let under = self.surface_under(location);
                pointer.motion(
                    self,
                    under,
                    &MotionEvent {
                        location,
                        serial: SERIAL_COUNTER.next_serial(),
                        time: event.time_msec(),
                    },
                );
                pointer.frame(self);
            }
            InputEvent::PointerMotionAbsolute { event, .. } => {
                let Some(pointer) = self.seat.get_pointer() else {
                    return;
                };
                let Some(output) = self.space.outputs().next() else {
                    return;
                };
                let Some(geo) = self.space.output_geometry(output) else {
                    return;
                };
                let location = event.position_transformed(geo.size) + geo.loc.to_f64();
                let under = self.surface_under(location);
                pointer.motion(
                    self,
                    under,
                    &MotionEvent {
                        location,
                        serial: SERIAL_COUNTER.next_serial(),
                        time: event.time_msec(),
                    },
                );
                pointer.frame(self);
            }
            InputEvent::PointerButton { event, .. } => {
                let Some(pointer) = self.seat.get_pointer() else {
                    return;
                };
                let serial = SERIAL_COUNTER.next_serial();
                if event.state() == smithay::backend::input::ButtonState::Pressed
                    && !pointer.is_grabbed()
                {
                    if let Some((window, _)) = self
                        .space
                        .element_under(pointer.current_location())
                        .map(|(window, location)| (window.clone(), location))
                    {
                        self.space.raise_element(&window, true);
                        if let Some(toplevel) = window.toplevel() {
                            if let Some(keyboard) = self.seat.get_keyboard() {
                                keyboard.set_focus(
                                    self,
                                    Some(toplevel.wl_surface().clone()),
                                    serial,
                                );
                            }
                            toplevel.send_pending_configure();
                        }
                    } else {
                        for window in self.space.elements() {
                            window.set_activated(false);
                            if let Some(toplevel) = window.toplevel() {
                                toplevel.send_pending_configure();
                            }
                        }
                        if let Some(keyboard) = self.seat.get_keyboard() {
                            keyboard.set_focus(self, Option::<WlSurface>::None, serial);
                        }
                    }
                }
                pointer.button(
                    self,
                    &ButtonEvent {
                        button: event.button_code(),
                        state: event.state(),
                        serial,
                        time: event.time_msec(),
                    },
                );
                pointer.frame(self);
            }
            InputEvent::PointerAxis { event, .. } => {
                let Some(pointer) = self.seat.get_pointer() else {
                    return;
                };
                let source = event.source();
                let horizontal = event
                    .amount(smithay::backend::input::Axis::Horizontal)
                    .unwrap_or(0.0);
                let vertical = event
                    .amount(smithay::backend::input::Axis::Vertical)
                    .unwrap_or(0.0);
                let mut frame = AxisFrame::new(event.time_msec()).source(source);
                if horizontal != 0.0 {
                    frame = frame.value(smithay::backend::input::Axis::Horizontal, horizontal);
                }
                if vertical != 0.0 {
                    frame = frame.value(smithay::backend::input::Axis::Vertical, vertical);
                }
                pointer.axis(self, frame);
                pointer.frame(self);
            }
            _ => {}
        }
    }
}

impl CompositorHandler for HitechShell {
    fn compositor_state(&mut self) -> &mut CompositorState {
        &mut self.compositor_state
    }

    fn client_compositor_state<'a>(&self, client: &'a Client) -> &'a CompositorClientState {
        &client
            .get_data::<ClientState>()
            .expect("client without compositor state")
            .compositor_state
    }

    fn commit(&mut self, surface: &WlSurface) {
        smithay::backend::renderer::utils::on_commit_buffer_handler::<Self>(surface);
        if !is_sync_subsurface(surface) {
            let mut root = surface.clone();
            while let Some(parent) = get_parent(&root) {
                root = parent;
            }
            if let Some(window) = self
                .space
                .elements()
                .find(|window| window.toplevel().map(|t| t.wl_surface()) == Some(&root))
            {
                window.on_commit();
            }
        }

        if let Some(window) = self
            .space
            .elements()
            .find(|window| window.toplevel().map(|t| t.wl_surface()) == Some(surface))
            .cloned()
        {
            let initial_configure_sent =
                smithay::wayland::compositor::with_states(surface, |states| {
                    states
                        .data_map
                        .get::<XdgToplevelSurfaceData>()
                        .expect("xdg toplevel without data")
                        .lock()
                        .expect("poisoned xdg toplevel data")
                        .initial_configure_sent
                });
            if !initial_configure_sent {
                if let Some(toplevel) = window.toplevel() {
                    let _ = toplevel.send_configure();
                }
            }
        }

        self.popups.commit(surface);
        if let Some(PopupKind::Xdg(popup)) = self.popups.find_popup(surface) {
            if !popup.is_initial_configure_sent() {
                let _ = popup.send_configure();
            }
        }

        self.queue_redraw();
    }
}

/// Build and run the compositor until the event loop is stopped.
pub fn run_compositor(config: &Config) -> Result<(), CompositorError> {
    let _ = tracing_subscriber::fmt()
        .with_env_filter(
            tracing_subscriber::EnvFilter::try_from_default_env()
                .unwrap_or_else(|_| tracing_subscriber::EnvFilter::new("info")),
        )
        .try_init();

    let mut event_loop: EventLoop<'static, HitechShell> =
        EventLoop::try_new().map_err(|e| CompositorError::Loop(e.to_string()))?;

    let display: Display<HitechShell> =
        Display::new().map_err(|e| CompositorError::Display(e.to_string()))?;

    let (session, notifier) =
        LibSeatSession::new().map_err(|e| CompositorError::Session(e.to_string()))?;

    let mut state = HitechShell::new(config.clone(), display, &mut event_loop, session, notifier)?;
    event_loop
        .run(None, &mut state, |_| {
            // The compositor runs until LoopSignal::stop() is called.
        })
        .map_err(|e| CompositorError::Loop(e.to_string()))?;

    Ok(())
}

/// Public compositor handle used by `main.rs`.
pub struct Compositor {
    config: Config,
}

impl Compositor {
    pub fn new(config: Config) -> Self {
        Self { config }
    }

    pub fn config(&self) -> &Config {
        &self.config
    }

    pub fn run(&self) {
        if let Err(e) = run_compositor(&self.config) {
            error!("compositor fatal: {e}");
            std::process::exit(1);
        }
    }
}

//! Config contract shared by the NixOS module and the Wayland shell.
//!
//! The NixOS module renders `/etc/hitechos/hitech-shell.toml` (or sets
//! `HITECH_SHELL_LAYOUT`) from the same enum below. The shell loads the same
//! value; `services.hitech-shell.layout` is therefore the single source of
//! truth in the installed system.

use std::env;

/// Layout profile exposed as the NixOS option
/// `services.hitech-shell.layout`.
#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum Layout {
    /// Windows-style bottom taskbar + start menu, plus an Android-style
    /// quick-settings panel.
    Hybrid,
    /// macOS-style bottom dock + top menu bar.
    Mac,
    /// Windows-style taskbar and start menu only.
    Windows,
}

impl Layout {
    pub fn parse(value: &str) -> Option<Self> {
        match value.trim().to_ascii_lowercase().as_str() {
            "hybrid" => Some(Self::Hybrid),
            "mac" | "macos" => Some(Self::Mac),
            "windows" | "win" => Some(Self::Windows),
            _ => None,
        }
    }

    pub fn as_str(self) -> &'static str {
        match self {
            Self::Hybrid => "hybrid",
            Self::Mac => "mac",
            Self::Windows => "windows",
        }
    }
}

#[derive(Clone, Debug)]
pub struct Config {
    pub layout: Layout,
    pub outputs: Vec<String>,
    pub source: String,
}

impl Config {
    /// Reads the environment set by the NixOS module/session.
    ///
    /// `HITECH_SHELL_LAYOUT` is authoritative; `HITECH_SHELL_CONFIG` points at
    /// the generated TOML file for later phases (multi-output settings,
    /// theme, widget allow-list). No parser dependency is pulled in yet.
    pub fn from_env() -> Self {
        let source =
            env::var("HITECH_SHELL_CONFIG").unwrap_or_else(|_| "<environment>".to_string());

        let layout = parse_layout(env::var("HITECH_SHELL_LAYOUT").ok().as_deref());
        let outputs = parse_outputs(env::var("HITECH_SHELL_OUTPUTS").ok().as_deref());

        Self {
            layout,
            outputs,
            source,
        }
    }
}

/// Parses `HITECH_SHELL_LAYOUT`, defaulting to the hybrid profile.
pub fn parse_layout(value: Option<&str>) -> Layout {
    value.and_then(Layout::parse).unwrap_or(Layout::Hybrid)
}

/// Parses `HITECH_SHELL_OUTPUTS` (comma-separated), defaulting to all outputs.
///
/// Empty entries are dropped so `"HDMI-A-1,,DP-1,"` yields two outputs and an
/// empty string falls back to `*`.
pub fn parse_outputs(value: Option<&str>) -> Vec<String> {
    let outputs: Vec<String> = value
        .unwrap_or("*")
        .split(',')
        .map(str::trim)
        .filter(|item| !item.is_empty())
        .map(ToOwned::to_owned)
        .collect();

    if outputs.is_empty() {
        vec!["*".to_string()]
    } else {
        outputs
    }
}

//! Layout profiles.
//!
//! Every profile is a pure description of screen regions and their anchor
//! semantics. The future Smithay layer-shell code consumes this table; keeping
//! it dependency-free means the NixOS module, tests and the compositor share a
//! single Rust enum with no UI toolkit lock-in.

use crate::config::Layout;

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum Anchor {
    Top,
    Bottom,
    Left,
    Right,
    Fullscreen,
}

impl Anchor {
    pub fn as_str(self) -> &'static str {
        match self {
            Self::Top => "top",
            Self::Bottom => "bottom",
            Self::Left => "left",
            Self::Right => "right",
            Self::Fullscreen => "fullscreen",
        }
    }
}

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum Layer {
    Background,
    Bottom,
    Top,
    Overlay,
}

impl Layer {
    pub fn as_str(self) -> &'static str {

// HiTech-OS ORION-X installer core.
//
// A small, dependency-free Rust HTTP/API server. The operator-facing UI is a
// TypeScript/HTML/SVG cockpit served from HITECH_WEB_ROOT (installed by the Nix
// package). Privileged work stays in this process: disks are inspected with
// lsblk, partitions are mounted explicitly, and the target is installed with
// nixos-install. Every mutating operation is triggered by an explicit POST.

fn main() {
    let port: u16 = env::var("HITECH_INSTALLER_PORT")
        .ok()
        .and_then(|p| p.parse().ok())
        .unwrap_or(17870);
    let web_root = web_root();
    let state = Arc::new(Mutex::new(InstallStatus {
        status: "Flight control online. Waiting for operator.".to_string(),
        ..InstallStatus::default()
    }));

    let listener = match TcpListener::bind(("127.0.0.1", port)) {
        Ok(l) => l,
        Err(e) => {
            eprintln!("cannot bind 127.0.0.1:{port}: {e}");
            std::process::exit(1);
        }
    };
    let url = format!("http://127.0.0.1:{port}/");
    println!("ORION-X cockpit listening on {url}");
    println!("web root: {}", web_root.display());

    // Browser launch is best-effort: the API remains reachable even if the
    // kiosk browser is missing or crashes.
    thread::spawn(move || {
        if let Err(e) = launch_browser(&url) {
            eprintln!("browser launch failed: {e}");
            eprintln!("open {url} manually");
        }
    });

    for stream in listener.incoming() {
        match stream {
            Ok(stream) => {
                let state = state.clone();
                let root = web_root.clone();
                thread::spawn(move || handle_client(stream, state, root));
            }
            Err(e) => eprintln!("connection error: {e}"),
        }
    }
}

fn web_root() -> PathBuf {
    if let Ok(p) = env::var("HITECH_WEB_ROOT") {
        return PathBuf::from(p);
    }
    if Path::new("frontend/index.html").exists() {
        return PathBuf::from("frontend");
    }
    PathBuf::from("share/hitech-installer/www")
}

fn handle_client(mut stream: TcpStream, state: Arc<Mutex<InstallStatus>>, root: PathBuf) {
    let _ = stream.set_read_timeout(Some(Duration::from_secs(8)));
    let request = match read_request(&mut stream) {
        Ok(r) => r,
        Err(e) => {
            let _ = send_text(&mut stream, 400, "bad request", &e);
            return;
        }
    };

    if request.path == "/api/status" && request.method == "GET" {
        let body = status_json(&state);
        let _ = send_json(&mut stream, 200, &body);
        return;
    }
    if request.path == "/api/network" && request.method == "GET" {
        let online = network_online();
        let _ = send_json(&mut stream, 200, &format!("{{\"online\":{}}}", online));
        return;
    }
    if request.path == "/api/disks" && request.method == "GET" {
        match scan_disks() {
            Ok(disks) => {
                let body = format!(
                    "{{\"disks\":[{}]}}",
                    disks.iter().map(disk_to_json).collect::<Vec<_>>().join(",")
                );
                let _ = send_json(&mut stream, 200, &body);
            }
            Err(e) => send_api_error(&mut stream, &e),
        }
        return;
    }
    if request.path == "/api/wifi/scan" && request.method == "POST" {
        match scan_wifi() {
            Ok(networks) => {
                let body = format!(
                    "{{\"networks\":[{}]}}",
                    networks
                        .iter()
                        .map(wifi_to_json)
                        .collect::<Vec<_>>()
                        .join(",")
                );
                let _ = send_json(&mut stream, 200, &body);
            }
            Err(e) => send_api_error(&mut stream, &e),
        }
        return;
    }
    if request.path == "/api/wifi/connect" && request.method == "POST" {
        let form = parse_form(&request.body);
        let ssid = form.get("ssid").cloned().unwrap_or_default();
        let password = form.get("password").cloned().unwrap_or_default();
        if ssid.is_empty() {
            send_api_error(&mut stream, "missing SSID");
            return;
        }
        match connect_wifi(&ssid, &password) {
            Ok(()) => {
                let _ = send_json(&mut stream, 200, "{\"ok\":true}");
            }
            Err(e) => send_api_error(&mut stream, &e),
        }
        return;
    }
    if request.path == "/api/gparted" && request.method == "POST" {
        match launch_detached("sudo", &["-E", "gparted"]) {
            Ok(()) => {
                let _ = send_json(&mut stream, 200, "{\"ok\":true}");
            }
            Err(e) => send_api_error(&mut stream, &e),
        }
        return;
    }
    if request.path == "/api/terminal" && request.method == "POST" {
        match launch_terminal() {
            Ok(()) => {
                let _ = send_json(&mut stream, 200, "{\"ok\":true}");
            }
            Err(e) => send_api_error(&mut stream, &e),
        }
        return;
    }
    if request.path == "/api/reboot" && request.method == "POST" {
        match launch_detached("sudo", &["reboot"]) {
            Ok(()) => {
                let _ = send_json(&mut stream, 200, "{\"ok\":true}");
            }
            Err(e) => send_api_error(&mut stream, &e),
        }
        return;
    }
    if request.path == "/api/install" && request.method == "POST" {
        let form = parse_form(&request.body);
        let req = InstallRequest {
            mode: form.get("mode").cloned().unwrap_or_default(),
            hostname: form.get("hostname").cloned().unwrap_or_default(),
            username: form.get("username").cloned().unwrap_or_default(),
            password: form.get("password").cloned().unwrap_or_default(),
            desktop: form.get("desktop").cloned().unwrap_or_default(),
            disk: form.get("disk").cloned().unwrap_or_default(),
            root_partition: form.get("root_partition").cloned().unwrap_or_default(),
            efi_partition: form.get("efi_partition").cloned().unwrap_or_default(),
        };
        if let Err(e) = validate_request(&req) {
            send_api_error(&mut stream, &e);
            return;
        }
        {
            let mut st = state
                .lock()
                .unwrap_or_else(|poisoned| poisoned.into_inner());
            if st.running {
                send_api_error(&mut stream, "installation already running");
                return;
            }
            st.running = true;
            st.success = false;
            st.progress = 0;
            st.status = "Ignition sequence started.".to_string();
            st.logs.clear();
        }
        let state2 = state.clone();
        thread::spawn(move || {
            let result = install_steps(&req, &state2);
            let mut st = state2
                .lock()
                .unwrap_or_else(|poisoned| poisoned.into_inner());
            st.running = false;
            st.success = result.is_ok();
            match result {
                Ok(()) => {
                    st.progress = 100;
                    st.status = "Installation complete. Target is ready to reboot.".to_string();
                    append_log_locked(&mut st, "== installation complete ==");
                }
                Err(e) => {
                    st.status = format!("Installation failed: {e}");
                    append_log_locked(&mut st, &format!("ERROR: {e}"));
                }
            }
        });
        let _ = send_json(&mut stream, 200, "{\"ok\":true}");
        return;
    }

    if request.path == "/" || request.path == "/index.html" {
        serve_static(&mut stream, &root, "index.html");
    } else if request.path == "/favicon.ico" {
        let _ = send_text(&mut stream, 204, "image/x-icon", "");
    } else {
        let rel = request.path.trim_start_matches('/');
        serve_static(&mut stream, &root, rel);
    }
}

fn read_request(stream: &mut TcpStream) -> Result<Request, String> {
    let mut buf = Vec::new();
    let mut chunk = [0u8; 4096];
    let header_end;
    loop {
        let n = stream.read(&mut chunk).map_err(|e| e.to_string())?;
        if n == 0 {
            return Err("empty request".to_string());
        }
        buf.extend_from_slice(&chunk[..n]);
        if let Some(pos) = find_bytes(&buf, b"\r\n\r\n") {
            header_end = pos + 4;
            break;
        }
        if buf.len() > 64 * 1024 {
            return Err("headers too large".to_string());
        }
    }
    let head = String::from_utf8_lossy(&buf[..header_end]).to_string();
    let mut lines = head.split("\r\n");
    let request_line = lines.next().unwrap_or("");
    let mut parts = request_line.split_whitespace();
    let method = parts.next().unwrap_or("").to_string();
    let target = parts.next().unwrap_or("/").to_string();
    let path = target.split('?').next().unwrap_or("/").to_string();
    let mut content_length = 0usize;
    for line in lines {
        if let Some((k, v)) = line.split_once(':') {
            if k.eq_ignore_ascii_case("content-length") {
                content_length = v.trim().parse().unwrap_or(0);
            }
        }
    }
    if content_length > 2 * 1024 * 1024 {
        return Err("body too large".to_string());
    }
    let mut body = buf[header_end..].to_vec();
    while body.len() < content_length {
        let n = stream.read(&mut chunk).map_err(|e| e.to_string())?;
        if n == 0 {
            break;
        }
        body.extend_from_slice(&chunk[..n]);
    }
    if body.len() > content_length {
        body.truncate(content_length);
    }
    Ok(Request {
        method,
        path,
        body: String::from_utf8_lossy(&body).to_string(),
    })
}

struct Request {
    method: String,
    path: String,
    body: String,
}

fn find_bytes(haystack: &[u8], needle: &[u8]) -> Option<usize> {
    haystack.windows(needle.len()).position(|w| w == needle)
}

fn send_response(
    stream: &mut TcpStream,
    status: u16,
    content_type: &str,
    body: &[u8],
) -> std::io::Result<()> {
    let reason = match status {
        200 => "OK",
        204 => "No Content",
        400 => "Bad Request",
        404 => "Not Found",
        500 => "Internal Server Error",
        _ => "OK",
    };
    let head = format!(
        "HTTP/1.1 {status} {reason}\r\nContent-Type: {content_type}\r\nContent-Length: {}\r\nCache-Control: no-store\r\nConnection: close\r\n\r\n",
        body.len()
    );
    stream.write_all(head.as_bytes())?;
    stream.write_all(body)?;
    stream.flush()
}

fn send_json(stream: &mut TcpStream, status: u16, body: &str) -> std::io::Result<()> {
    send_response(
        stream,
        status,
        "application/json; charset=utf-8",
        body.as_bytes(),
    )
}

fn send_text(
    stream: &mut TcpStream,
    status: u16,
    content_type: &str,
    body: &str,
) -> std::io::Result<()> {
    send_response(stream, status, content_type, body.as_bytes())
}

fn send_api_error(stream: &mut TcpStream, message: &str) {
    let body = format!("{{\"error\":\"{}\"}}", json_escape(message));
    let _ = send_json(stream, 400, &body);
}

fn serve_static(stream: &mut TcpStream, root: &Path, rel: &str) {
    if rel.contains("..") {
        let _ = send_text(stream, 404, "text/plain; charset=utf-8", "not found");
        return;
    }
    let path = root.join(rel);
    let canonical_root = match root.canonicalize() {
        Ok(p) => p,
        Err(_) => root.to_path_buf(),
    };
    let canonical_path = match path.canonicalize() {
        Ok(p) if p.starts_with(&canonical_root) => p,
        _ => {
            let _ = send_text(stream, 404, "text/plain; charset=utf-8", "not found");
            return;
        }
    };
    match fs::read(&canonical_path) {
        Ok(bytes) => {
            let mime = match canonical_path
                .extension()
                .and_then(|e| e.to_str())
                .unwrap_or("")
            {
                "html" => "text/html; charset=utf-8",
                "css" => "text/css; charset=utf-8",
                "js" => "text/javascript; charset=utf-8",
                "svg" => "image/svg+xml",
                "png" => "image/png",
                _ => "application/octet-stream",
            };
            let _ = send_response(stream, 200, mime, &bytes);
        }
        Err(_) => {
            let _ = send_text(stream, 404, "text/plain; charset=utf-8", "not found");
        }
    }
}

fn status_json(state: &Arc<Mutex<InstallStatus>>) -> String {
    // Probe the network before taking the status lock. A slow `nmcli` call
    // must never block the install thread's progress updates.
    let online = network_online();
    let mut st = state
        .lock()
        .unwrap_or_else(|poisoned| poisoned.into_inner());
    st.network_online = online;
    format!(
        "{{\"running\":{},\"success\":{},\"progress\":{},\"status\":\"{}\",\"logs\":\"{}\",\"network_online\":{}}}",
        st.running, st.success, st.progress, json_escape(&st.status), json_escape(&st.logs), st.network_online
    )
}

fn set_status(state: &Arc<Mutex<InstallStatus>>, progress: u16, status: &str) {
    let mut st = state
        .lock()
        .unwrap_or_else(|poisoned| poisoned.into_inner());
    st.progress = progress.min(100);
    st.status = status.to_string();
    append_log_locked(&mut st, &format!("[{progress:>3}%] {status}"));
}

fn append_log(state: &Arc<Mutex<InstallStatus>>, line: &str) {
    let mut st = state
        .lock()
        .unwrap_or_else(|poisoned| poisoned.into_inner());
    append_log_locked(&mut st, line);
}

fn append_log_locked(st: &mut InstallStatus, line: &str) {
    if !st.logs.is_empty() {
        st.logs.push('\n');
    }
    st.logs.push_str(line);
    if st.logs.len() > INSTALL_LOG_LIMIT {
        let mut cut = st.logs.len() - INSTALL_LOG_LIMIT;
        while !st.logs.is_char_boundary(cut) {
            cut += 1;
        }
        let keep = st.logs[cut..]
            .find('\n')
            .map(|i| cut + i + 1)
            .unwrap_or(cut);
        st.logs = st.logs[keep..].to_string();
    }
}

fn validate_request(req: &InstallRequest) -> Result<(), String> {
    if !["replace", "erase"].contains(&req.mode.as_str()) {
        return Err("invalid mode".to_string());
    }
    if !["orion", "plasma6", "gnome", "xfce"].contains(&req.desktop.as_str()) {
        return Err("invalid desktop".to_string());
    }
    if !valid_hostname(&req.hostname) {
        return Err("invalid hostname".to_string());
    }
    if !valid_username(&req.username) {
        return Err("invalid username".to_string());
    }
    if req.password.chars().count() < 4 {
        return Err("password must contain at least 4 characters".to_string());
    }
    if req.mode == "replace" {
        if !req.root_partition.starts_with("/dev/") {
            return Err("invalid root partition".to_string());
        }
        if !req.efi_partition.starts_with("/dev/") {
            return Err("invalid EFI partition".to_string());
        }
        if req.root_partition == req.efi_partition {
            return Err("root and EFI partitions must differ".to_string());
        }
    } else if !req.disk.starts_with("/dev/") {
        return Err("invalid target disk".to_string());
    }
    Ok(())
}

fn valid_hostname(s: &str) -> bool {
    let mut chars = s.chars();
    match chars.next() {
        Some(c) if c.is_ascii_alphanumeric() => {}
        _ => return false,
    }
    s.len() <= 63 && s.chars().all(|c| c.is_ascii_alphanumeric() || c == '-')
}

fn valid_username(s: &str) -> bool {
    let mut chars = s.chars();
    match chars.next() {
        Some(c) if c.is_ascii_lowercase() || c == '_' => {}
        _ => return false,
    }
    s.len() <= 32
        && s.chars()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_' || c == '-')
}

fn nix_escape(s: &str) -> String {
    format!(
        "\"{}\"",
        s.replace('\\', "\\\\")
            .replace('"', "\\\"")
            .replace("${", "\\${")
    )
}

fn install_steps(req: &InstallRequest, state: &Arc<Mutex<InstallStatus>>) -> Result<(), String> {
    set_status(state, 2, "Preparing target storage…");
    if req.mode == "replace" {
        prepare_existing(&req.root_partition, &req.efi_partition, state)?;
    } else {
        prepare_erase(&req.disk, state)?;
    }

    set_status(state, 24, "Generating hardware configuration…");
    run_checked(state, "nixos-generate-config", &["--root", "/mnt"])?;

    set_status(state, 28, "Provisioning the pinned HiTech-OS profile…");
    fetch_profile(state)?;

    set_status(
        state,
        34,
        "Writing target identity and desktop configuration…",
    );
    let hash = hash_password(&req.password)?;
    let config = target_config(req, &hash);
    root_mkdir(state, "/mnt/etc/nixos")?;
    root_write(state, "/mnt/etc/nixos/configuration.nix", &config)?;
    verify_target_profile()?;

    let fallback = env::var("HITECH_ALLOW_SOURCE_FALLBACK")
        .map(|v| v != "false" && v != "0")
        .unwrap_or(true);
    let mut last_error = String::new();
    for attempt in 1..=3 {
        set_status(
            state,
            38 + (attempt as u16 - 1) * 4,
            &format!("Installing system closure (attempt {attempt}/3)…"),
        );
        append_log(state, &format!("== nixos-install attempt {attempt}/3 =="));
        let args = [
            "--root",
            "/mnt",
            "--no-root-passwd",
            "--option",
            "fallback",
            if fallback { "true" } else { "false" },
            "--option",
            "substitute",
            "true",
        ];
        match run_stream(state, "nixos-install", &args) {
            Ok(()) => {
                last_error.clear();
                break;
            }
            Err(e) => {
                last_error = e.clone();
                append_log(state, &format!("attempt {attempt} failed: {e}"));
                if attempt < 3 {
                    set_status(
                        state,
                        40,
                        &format!("Installer I/O/network retry {}/3…", attempt + 1),
                    );
                    thread::sleep(Duration::from_secs(4));
                }
            }
        }
    }
    if !last_error.is_empty() {
        return Err(format!(
            "nixos-install failed after 3 attempts: {last_error}"
        ));
    }

    set_status(state, 96, "Flushing caches and unmounting target…");
    let _ = run_checked(state, "sync", &[]);
    let _ = run_checked(state, "umount", &["-R", "/mnt"]);
    Ok(())
}

/// /mnt is root-owned once the target is mounted, so even plain file writes
/// must be performed through the privileged path.
fn root_mkdir(state: &Arc<Mutex<InstallStatus>>, path: &str) -> Result<(), String> {
    run_checked(state, "mkdir", &["-p", path])
}

fn root_write(state: &Arc<Mutex<InstallStatus>>, path: &str, contents: &str) -> Result<(), String> {
    let tmp = format!("/tmp/hitech-target-{}.tmp", std::process::id());
    fs::write(&tmp, contents).map_err(|e| format!("cannot stage {path}: {e}"))?;
    let result = run_checked(state, "install", &["-m", "0644", &tmp, path]);
    let _ = fs::remove_file(&tmp);
    result
}

fn prepare_existing(
    root: &str,
    efi: &str,
    state: &Arc<Mutex<InstallStatus>>,
) -> Result<(), String> {
    set_status(state, 5, "Unmounting any stale target mounts…");
    unmount_path(root, state);
    unmount_path(efi, state);
    let _ = run_checked(state, "udevadm", &["settle"]);

    set_status(state, 10, "Formatting the selected root partition…");
    run_checked(state, "wipefs", &["-a", root])?;
    run_checked(state, "mkfs.ext4", &["-F", "-L", "nixos", root])?;

    set_status(
        state,
        14,
        "Mounting root and preserving the existing EFI partition…",
    );
    run_checked(state, "mkdir", &["-p", "/mnt"])?;
    run_checked(state, "mount", &[root, "/mnt"])?;
    run_checked(state, "mkdir", &["-p", "/mnt/boot"])?;
    run_checked(state, "mount", &[efi, "/mnt/boot"])?;
    Ok(())
}

fn prepare_erase(disk: &str, state: &Arc<Mutex<InstallStatus>>) -> Result<(), String> {
    set_status(state, 5, "Unmounting all partitions on the target disk…");
    if let Ok(disks) = scan_disks() {
        if let Some(d) = disks.iter().find(|d| d.path == disk) {
            for p in &d.partitions {
                unmount_path(&p.path, state);
            }
        }
    }
    unmount_path(disk, state);
    let _ = run_checked(state, "udevadm", &["settle"]);

    set_status(state, 8, "Erasing disk and writing GPT…");
    run_checked(state, "wipefs", &["-af", disk])?;
    run_checked(state, "parted", &["--script", disk, "mklabel", "gpt"])?;
    run_checked(
        state,
        "parted",
        &["--script", disk, "mkpart", "ESP", "fat32", "1MiB", "513MiB"],
    )?;
    run_checked(
        state,
        "parted",
        &["--script", disk, "set", "1", "esp", "on"],
    )?;
    run_checked(
        state,
        "parted",
        &["--script", disk, "mkpart", "root", "ext4", "513MiB", "100%"],
    )?;
    let _ = run_checked(state, "partprobe", &[disk]);
    let _ = run_checked(state, "udevadm", &["settle"]);
    thread::sleep(Duration::from_millis(500));

    let efi = partition_path(disk, 1);
    let root = partition_path(disk, 2);
    set_status(state, 12, "Formatting EFI and root…");
    run_checked(state, "mkfs.fat", &["-F32", "-n", "HITECH-EFI", &efi])?;
    run_checked(state, "mkfs.ext4", &["-F", "-L", "nixos", &root])?;
    set_status(state, 16, "Mounting target…");
    run_checked(state, "mkdir", &["-p", "/mnt"])?;
    run_checked(state, "mount", &[&root, "/mnt"])?;
    run_checked(state, "mkdir", &["-p", "/mnt/boot"])?;
    run_checked(state, "mount", &[&efi, "/mnt/boot"])?;
    Ok(())
}

fn partition_path(disk: &str, n: u32) -> String {
    if disk
        .chars()
        .last()
        .map(|c| c.is_ascii_digit())
        .unwrap_or(false)
    {
        format!("{disk}p{n}")
    } else {
        format!("{disk}{n}")
    }
}

fn unmount_path(path: &str, state: &Arc<Mutex<InstallStatus>>) {
    let output = Command::new("findmnt")
        .args(["-rn", "-S", path, "-o", "TARGET"])
        .output();
    let Ok(out) = output else {
        return;
    };
    for target in String::from_utf8_lossy(&out.stdout)
        .lines()
        .map(str::trim)
        .filter(|s| !s.is_empty())
    {
        if target == "/"
            || target == "/nix"
            || target == "/nix/store"
            || target.starts_with("/sys")
            || target.starts_with("/proc")
            || target == "/boot"
        {
            append_log(
                state,
                &format!("WARN: refusing to unmount live mount {target} ({path})"),
            );
            continue;
        }
        if target.starts_with("/run/media/")
            || target.starts_with("/media/")
            || target == "/mnt"
            || target.starts_with("/mnt/")
        {
            append_log(state, &format!("unmount {target} ({path})"));
            let _ = privileged("umount").args(["-l", target]).output();
        }
    }
}

fn resolve_repo_url(value: Option<&str>) -> String {
    value
        .map(str::trim)
        .filter(|value| !value.is_empty())
        .unwrap_or(DEFAULT_REPO_URL)
        .to_string()
}

fn resolve_repo_ref(value: Option<&str>) -> String {
    value
        .map(str::trim)
        .filter(|value| !value.is_empty())
        .unwrap_or("main")
        .to_string()
}

fn profile_is_complete(repo: &Path) -> bool {
    ["flake.nix", "os/installation.nix", "os/common.nix"]
        .iter()
        .all(|rel| repo.join(rel).is_file())
}

/// Copy the complete live-ISO source bundle into the target.
///
/// The runner is injected so the exact command sequence is unit-testable
/// without requiring root, sudo, or a real /mnt.
fn provision_profile_from_source<F>(source: &Path, dest: &Path, mut runner: F) -> Result<(), String>
where
    F: FnMut(&str, &[&str]) -> Result<(), String>,
{
    if !source.is_dir() {
        return Err(format!(
            "offline source bundle is missing: {}",
            source.display()
        ));
    }
    let source_arg = format!("{}/.", source.display());
    let dest_arg = dest.display().to_string();
    runner("rm", &["-rf", &dest_arg])?;
    runner("mkdir", &["-p", &dest_arg])?;
    runner("cp", &["-rL", &source_arg, &dest_arg])?;
    runner("chmod", &["-R", "u+w", &dest_arg])?;
    // A copied .git directory is never part of the installed profile and may
    // contain absolute worktree paths from the ISO build. Keep the target
    // tree immutable in intent, but writable for nixos-install.
    let git_dir = dest.join(".git");
    if git_dir.exists() {
        runner("rm", &["-rf", &git_dir.display().to_string()])?;
    }
    if !profile_is_complete(dest) {
        return Err(format!(
            "offline source bundle is incomplete: {}",
            dest.display()
        ));
    }
    Ok(())
}

fn fetch_profile(state: &Arc<Mutex<InstallStatus>>) -> Result<(), String> {
    let url = resolve_repo_url(env::var("HITECH_REPO_URL").ok().as_deref());
    let refname = resolve_repo_ref(env::var("HITECH_REPO_REF").ok().as_deref());
    let dest = Path::new(TARGET_REPO_DIR);
    let source = Path::new(LIVE_SOURCE_DIR);

    // Offline-first: both ISOs embed the complete repository at
    // /etc/hitech-os-source. This path never depends on the network, a git
    // credential, or an empty environment variable.
    if source.is_dir() {
        append_log(
            state,
            &format!(
                "offline source bundle found: {} -> {}",
                source.display(),
                dest.display()
            ),
        );
        let result = provision_profile_from_source(source, dest, |program, args| {
            run_checked(state, program, args)
        });
        match result {
            Ok(()) => return Ok(()),
            Err(error) => append_log(
                state,
                &format!("WARN: offline source bundle unusable ({error}); falling back to git"),
            ),
        }
    }

    append_log(
        state,
        &format!(
            "fetching HiTech-OS profile {url} ({refname}) -> {}",
            dest.display()
        ),
    );
    if dest.exists() {
        run_checked(state, "rm", &["-rf", TARGET_REPO_DIR])?;
    }
    for args in git_fetch_plan(TARGET_REPO_DIR, &url, &refname) {
        let refs: Vec<&str> = args.iter().map(String::as_str).collect();
        run_checked(state, "git", &refs)?;
    }
    if !profile_is_complete(dest) {
        return Err(format!(
            "HiTech-OS profile is incomplete after git fetch: {}/os/installation.nix",
            TARGET_REPO_DIR
        ));
    }
    Ok(())
}

fn verify_target_profile() -> Result<(), String> {
    let repo = Path::new(TARGET_REPO_DIR);
    let installation = repo.join("os/installation.nix");
    if !installation.is_file() {
        return Err(format!(
            "pre-flight failed: required path '{}' does not exist",
            installation.display()
        ));
    }
    let config_path = Path::new("/mnt/etc/nixos/configuration.nix");
    let config = fs::read_to_string(config_path)
        .map_err(|e| format!("pre-flight: cannot read {}: {e}", config_path.display()))?;
    if !config.contains("./hitech-os/os/installation.nix") {
        return Err(format!(
            "pre-flight failed: {} does not import ./hitech-os/os/installation.nix",
            config_path.display()
        ));
    }
    Ok(())
}

/// Builds the git command sequence that fetches one profile revision.
///
/// `git clone --branch <sha>` only accepts branch/tag names, so a release ISO
/// that bakes `self.rev` (a full commit SHA) used to fail at install time. The
/// sequence below initialises the target, fetches exactly one revision (branch,
/// tag or commit) and detaches onto `FETCH_HEAD`; it works for all three and
/// keeps the clone shallow.
fn git_fetch_plan(dest: &str, url: &str, reference: &str) -> Vec<Vec<String>> {
    let reference = reference.trim();
    let fetch_ref = if reference.is_empty() {
        "HEAD"
    } else {
        reference
    };
    vec![
        vec!["init".into(), "-q".into(), dest.into()],
        vec![
            "-C".into(),
            dest.into(),
            "remote".into(),
            "add".into(),
            "origin".into(),
            url.into(),
        ],
        vec![
            "-C".into(),
            dest.into(),
            "fetch".into(),
            "--depth".into(),
            "1".into(),
            "origin".into(),
            fetch_ref.into(),
        ],
        vec![
            "-C".into(),
            dest.into(),
            "checkout".into(),
            "--detach".into(),
            "FETCH_HEAD".into(),
        ],
    ]
}

fn hash_password(password: &str) -> Result<String, String> {
    if let Ok(out) = Command::new("mkpasswd")
        .args(["-m", "sha-512", password])
        .output()
    {
        if out.status.success() {
            let hash = String::from_utf8_lossy(&out.stdout).trim().to_string();
            if hash.starts_with("$6$") {
                return Ok(hash);
            }
        }
    }
    let out = Command::new("openssl")
        .args(["passwd", "-6", password])
        .output()
        .map_err(|e| format!("cannot hash password: {e}"))?;
    if !out.status.success() {
        return Err("password hashing failed".to_string());
    }
    Ok(String::from_utf8_lossy(&out.stdout).trim().to_string())
}

fn target_config(req: &InstallRequest, password_hash: &str) -> String {
    let mut s = String::new();
    s.push_str("{ config, pkgs, lib, ... }:\n\n{\n  imports = [\n    ./hardware-configuration.nix\n    ./hitech-os/os/installation.nix\n  ];\n\n");
    s.push_str(&format!(
        "  networking.hostName = {};\n",
        nix_escape(&req.hostname)
    ));
    s.push_str("  networking.networkmanager.enable = true;\n");
    s.push_str("  time.timeZone = \"UTC\";\n");
    s.push_str(&format!(
        "  users.users.{} = {{\n",
        nix_escape(&req.username)
    ));
    s.push_str("    isNormalUser = true;\n");
    s.push_str(&format!(
        "    description = {};\n",
        nix_escape(&req.username)
    ));
    s.push_str(
        "    extraGroups = [ \"wheel\" \"networkmanager\" \"video\" \"render\" \"input\" ];\n",
    );
    s.push_str(&format!(
        "    initialHashedPassword = {};\n",
        nix_escape(password_hash)
    ));
    s.push_str("  };\n");
    s.push_str("  security.sudo.wheelNeedsPassword = true;\n");
    s.push_str("  security.polkit.enable = true;\n");
    s.push_str("  services.udisks2.enable = true;\n");
    s.push_str("  services.libinput.enable = true;\n");
    s.push_str("  services.xserver.enable = true;\n");
    s.push_str("  boot.loader.systemd-boot.enable = true;\n");
    s.push_str("  boot.loader.efi.canTouchEfiVariables = true;\n");
    s.push_str(&format!(
        "  system.stateVersion = {};\n",
        nix_escape(&env::var("HITECH_STATE_VERSION").unwrap_or_else(|_| "25.11".to_string()))
    ));
    s.push_str("  nix.settings.experimental-features = [ \"nix-command\" \"flakes\" ];\n");
    s.push_str("\n  # HiTech-Shell: always enabled when Orion desktop is selected.\n");
    s.push_str("  services.hitech-shell.enable = true;\n");
    match req.desktop.as_str() {
        "orion" => {
            s.push_str("  services.displayManager.sddm.enable = true;\n");
            s.push_str("  services.displayManager.sddm.wayland.enable = true;\n");
            s.push_str("  services.displayManager.defaultSession = \"hitech-shell\";\n");
            s.push_str("  services.displayManager.autoLogin.enable = true;\n");
            s.push_str(&format!(
                "  services.displayManager.autoLogin.user = {};\n",
                nix_escape(&req.username)
            ));
        }
        "plasma6" => {
            s.push_str("  services.desktopManager.plasma6.enable = true;\n");
            s.push_str("  services.displayManager.sddm.enable = true;\n");
        }
        "gnome" => {
            s.push_str("  services.desktopManager.gnome.enable = true;\n");
            s.push_str("  services.displayManager.gdm.enable = true;\n");
        }
        "xfce" => {
            s.push_str("  services.xserver.desktopManager.xfce.enable = true;\n");
            s.push_str("  services.xserver.displayManager.lightdm.enable = true;\n");
        }
        _ => {}
    }
    s.push_str("}\n");
    s
}

/// The installer runs as the unprivileged live user (Cage kiosk and the Live
/// GNOME session both start it without root), while partitioning, mounting and
/// nixos-install all need root. Privileged work goes through the live images'
/// NOPASSWD wheel policy; when the process is already root it runs directly.
fn is_root() -> bool {
    fs::read_to_string("/proc/self/status")
        .ok()
        .and_then(|s| {
            s.lines().find(|l| l.starts_with("Uid:")).and_then(|l| {
                l.split_whitespace()
                    .nth(1)
                    .and_then(|v| v.parse::<u32>().ok())
            })
        })
        .map(|uid| uid == 0)
        .unwrap_or(false)
}

fn privileged(program: &str) -> Command {
    if is_root() {
        Command::new(program)
    } else {
        let mut cmd = Command::new("sudo");
        cmd.args(["-n", "--"]).arg(program);
        cmd
    }
}

fn run_checked(
    state: &Arc<Mutex<InstallStatus>>,
    program: &str,
    args: &[&str],
) -> Result<(), String> {
    let shown = format!("{program} {}", args.join(" "));
    append_log(state, &format!("$ {shown}"));
    let out = privileged(program)
        .args(args)
        .output()
        .map_err(|e| format!("{program}: {e}"))?;
    if !out.stdout.is_empty() {
        for line in String::from_utf8_lossy(&out.stdout)
            .lines()
            .filter(|l| !l.trim().is_empty())
        {
            append_log(state, line);
        }
    }
    if !out.stderr.is_empty() {
        for line in String::from_utf8_lossy(&out.stderr)
            .lines()
            .filter(|l| !l.trim().is_empty())
        {
            append_log(state, line);
        }
    }
    if out.status.success() {
        Ok(())
    } else {
        Err(format!("{shown} exited with {}", out.status))
    }
}

fn run_stream(
    state: &Arc<Mutex<InstallStatus>>,
    program: &str,
    args: &[&str],
) -> Result<(), String> {
    append_log(state, &format!("$ {program} {}", args.join(" ")));
    let mut child = privileged(program)
        .args(args)
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .map_err(|e| format!("{program}: {e}"))?;
    let stdout = child.stdout.take();
    let stderr = child.stderr.take();
    let s1 = state.clone();
    let s2 = state.clone();
    let h1 = stdout.map(|out| {
        thread::spawn(move || {
            for line in BufReader::new(out).lines().map_while(Result::ok) {
                if !line.trim().is_empty() {
                    append_log(&s1, &line);
                }
            }
        })
    });
    let h2 = stderr.map(|err| {
        thread::spawn(move || {
            for line in BufReader::new(err).lines().map_while(Result::ok) {
                if !line.trim().is_empty() {
                    append_log(&s2, &line);
                }
            }
        })
    });
    let status = child.wait().map_err(|e| e.to_string())?;
    if let Some(h) = h1 {
        let _ = h.join();
    }
    if let Some(h) = h2 {
        let _ = h.join();
    }
    if status.success() {
        Ok(())
    } else {
        Err(format!("{program} exited with {status}"))
    }
}

fn run_capture(program: &str, args: &[&str]) -> Result<String, String> {
    let out = Command::new(program)
        .args(args)
        .output()
        .map_err(|e| format!("{program}: {e}"))?;
    if out.status.success() {
        Ok(String::from_utf8_lossy(&out.stdout).to_string())
    } else {
        Err(String::from_utf8_lossy(&out.stderr).trim().to_string())
    }
}

fn network_online() -> bool {
    // Bound each probe: the status endpoint is polled by the UI and must stay
    // responsive even when NetworkManager's D-Bus service is wedged.
    if let Ok(out) = Command::new("timeout")
        .args(["2", "ip", "route", "show", "default"])
        .output()
    {
        if out.status.success() && !out.stdout.is_empty() {
            return true;
        }
    }
    if let Ok(out) = Command::new("timeout")
        .args(["2", "nmcli", "-t", "-f", "STATE", "general"])
        .output()
    {
        if out.status.success() && String::from_utf8_lossy(&out.stdout).contains("connected") {
            return true;
        }
    }
    false
}

fn scan_wifi() -> Result<Vec<(String, String, String)>, String> {
    let output = run_capture(
        "nmcli",
        &["-t", "-f", "SSID,SIGNAL,SECURITY", "dev", "wifi", "list"],
    )?;
    let mut seen = Vec::new();
    let mut result = Vec::new();
    for line in output.lines() {
        let mut parts = line.splitn(3, ':');
        let ssid = parts.next().unwrap_or("").trim().to_string();
        let signal = parts.next().unwrap_or("").trim().to_string();
        let security = parts.next().unwrap_or("").trim().to_string();
        if ssid.is_empty() || ssid == "--" || seen.contains(&ssid) {
            continue;
        }
        seen.push(ssid.clone());
        result.push((ssid, signal, security));
        if result.len() >= 40 {
            break;
        }
    }
    Ok(result)
}

fn connect_wifi(ssid: &str, password: &str) -> Result<(), String> {
    if password.is_empty() {
        run_capture("nmcli", &["dev", "wifi", "connect", ssid])?;
    } else {
        run_capture(
            "nmcli",
            &["dev", "wifi", "connect", ssid, "password", password],
        )?;
    }
    Ok(())
}

fn launch_detached(program: &str, args: &[&str]) -> Result<(), String> {
    Command::new(program)
        .args(args)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .map(|_| ())
        .map_err(|e| format!("{program}: {e}"))
}

fn launch_terminal() -> Result<(), String> {
    for (program, args) in [
        ("foot", vec!["-e", "bash", "-l"]),
        ("kgx", vec!["--", "bash", "-l"]),
        ("gnome-terminal", vec!["--", "bash", "-l"]),
        ("xterm", vec!["-e", "bash", "-l"]),
    ] {
        if launch_detached(program, &args).is_ok() {
            return Ok(());
        }
    }
    Err("no terminal emulator found".to_string())
}

fn launch_browser(url: &str) -> Result<(), String> {
    let candidates = env::var("HITECH_BROWSER")
        .ok()
        .map(|b| vec![b])
        .unwrap_or_else(|| {
            vec![
                "chromium".to_string(),
                "chromium-browser".to_string(),
                "google-chrome".to_string(),
                "firefox".to_string(),
            ]
        });
    let wayland = env::var("WAYLAND_DISPLAY").is_ok();
    // A desktop session (GNOME on the Live ISO) has a window manager: an app
    // window that starts fullscreen gives the immersive cockpit while Alt+Tab
    // still reaches the rest of the desktop. With no session shell (Cage on the
    // NetInstaller) we take the whole seat in real kiosk mode instead.
    let desktop_session = env::var("XDG_CURRENT_DESKTOP")
        .map(|d| !d.trim().is_empty())
        .unwrap_or(false)
        && env::var("HITECH_FORCE_KIOSK")
            .map(|v| v == "0")
            .unwrap_or(true);
    let mut last = String::new();
    for name in candidates {
        let lower = name.to_lowercase();
        let mut cmd = Command::new(&name);
        if lower.contains("firefox") {
            cmd.args(["--kiosk", url]);
        } else {
            if desktop_session {
                // Fullscreen maximized window with --app for kiosk-style immersive,
                // but window decorations remain via the session shell so Alt+Tab works.
                cmd.arg("--app")
                    .arg(url)
                    .arg("--start-fullscreen")
                    .arg("--start-maximized");
            } else {
                cmd.arg("--kiosk").arg("--start-fullscreen").arg(url);
            }
            cmd.arg("--password-store=basic")
                .arg("--no-first-run")
                .arg("--no-default-browser-check")
                .arg("--disable-gpu")
                .env("LIBGL_ALWAYS_SOFTWARE", "1")
                .arg("--disable-dev-shm-usage")
                .arg("--window-size=1920,1080");
            if wayland {
                cmd.arg("--ozone-platform=wayland");
            }
        }
        cmd.stdin(Stdio::null())
            .stdout(Stdio::null())
            .stderr(Stdio::null());
        match cmd.spawn() {
            Ok(_) => return Ok(()),
            Err(e) => last = format!("{name}: {e}"),
        }
    }
    Err(if last.is_empty() {
        "no browser found".to_string()
    } else {
        last
    })
}

fn parse_form(body: &str) -> HashMap<String, String> {
    let mut result = HashMap::new();
    for pair in body.split('&') {
        let (k, v) = pair.split_once('=').unwrap_or((pair, ""));
        result.insert(percent_decode(k), percent_decode(v));
    }
    result
}

fn percent_decode(s: &str) -> String {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'+' => {
                out.push(b' ');
                i += 1;
            }
            b'%' if i + 2 < bytes.len() => {
                let hex = &s[i + 1..i + 3];
                if let Ok(v) = u8::from_str_radix(hex, 16) {
                    out.push(v);
                    i += 3;
                } else {
                    out.push(bytes[i]);
                    i += 1;
                }
            }
            b => {
                out.push(b);
                i += 1;
            }
        }
    }
    String::from_utf8_lossy(&out).to_string()
}

fn disk_to_json(d: &Disk) -> String {
    format!(
        "{{\"name\":\"{}\",\"path\":\"{}\",\"size_bytes\":{},\"size_human\":\"{}\",\"model\":\"{}\",\"partitions\":[{}]}}",
        json_escape(&d.name), json_escape(&d.path), d.size_bytes, json_escape(&d.size_human), json_escape(&d.model),
        d.partitions.iter().map(partition_to_json).collect::<Vec<_>>().join(",")
    )
}

fn partition_to_json(p: &Partition) -> String {
    format!(
        "{{\"name\":\"{}\",\"path\":\"{}\",\"size_bytes\":{},\"size_human\":\"{}\",\"fstype\":\"{}\",\"mountpoint\":\"{}\",\"label\":\"{}\",\"partlabel\":\"{}\",\"uuid\":\"{}\",\"parttype\":\"{}\",\"is_efi\":{}}}",
        json_escape(&p.name), json_escape(&p.path), p.size_bytes, json_escape(&p.size_human), json_escape(&p.fstype),
        json_escape(&p.mountpoint), json_escape(&p.label), json_escape(&p.partlabel), json_escape(&p.uuid),
        json_escape(&p.parttype), p.is_efi
    )
}

fn wifi_to_json(w: &(String, String, String)) -> String {

        "{{\"ssid\":\"{}\",\"signal\":\"{}\",\"security\":\"{}\"}}",

    )
}

fn scan_disks() -> Result<Vec<Disk>, String> {
    let out = Command::new("lsblk")
        .args([
            "-J",
            "-b",
            "-o",
            "NAME,PATH,SIZE,TYPE,FSTYPE,MOUNTPOINT,LABEL,PARTLABEL,MODEL,PARTTYPE,UUID",
        ])
        .output()
        .map_err(|e| format!("lsblk: {e}"))?;
    if !out.status.success() {
        return Err(String::from_utf8_lossy(&out.stderr).trim().to_string());
    }
    let root = JsonParser::new(&out.stdout).parse()?;
    let arr = match root {
        Json::Obj(_) => json_get(&root, "blockdevices")
            .cloned()
            .unwrap_or(Json::Arr(vec![])),
        other => other,
    };
    let items = match arr {
        Json::Arr(v) => v,
        _ => return Err("unexpected lsblk JSON".to_string()),
    };
    let mut disks = Vec::new();
    for item in items {
        if json_str(&item, "type") != "disk" {
            continue;
        }
        let path = json_str(&item, "path");
        if path.starts_with("/dev/loop")
            || path.starts_with("/dev/sr")
            || path.starts_with("/dev/zram")
            || path.starts_with("/dev/ram")
            || path.starts_with("/dev/fd")
        {
            continue;
        }
        let size = json_u64(&item, "size");
        if size < MIN_DISK_BYTES {
            continue;
        }
        let mut partitions = Vec::new();
        if let Some(Json::Arr(children)) = json_get(&item, "children") {
            for c in children {
                let psize = json_u64(c, "size");
                if psize == 0 {
                    continue;
                }
                let fstype = json_str(c, "fstype");
                let mountpoint = json_str(c, "mountpoint");
                let parttype = json_str(c, "parttype");
                let parttype_l = parttype.to_lowercase();
                let fstype_l = fstype.to_lowercase();
                let is_efi = fstype_l == "vfat"
                    || fstype_l == "fat"
                    || fstype_l == "fat32"
                    || parttype_l.contains("c12a7328-f81f-11d2-ba4b-00a0c93ec93b")
                    || mountpoint == "/boot"
                    || mountpoint == "/efi"
                    || mountpoint == "/boot/efi";
                partitions.push(Partition {
                    name: json_str(c, "name"),
                    path: json_str(c, "path"),
                    size_bytes: psize,
                    size_human: human_size(psize),
                    fstype,
                    mountpoint,
                    label: json_str(c, "label"),
                    partlabel: json_str(c, "partlabel"),
                    uuid: json_str(c, "uuid"),
                    parttype,
                    is_efi,
                });
            }
        }
        disks.push(Disk {
            name: json_str(&item, "name"),
            path,
            size_bytes: size,
            size_human: human_size(size),
            model: json_str(&item, "model").trim().to_string(),
            partitions,
        });
    }
    disks.sort_by(|a, b| a.path.cmp(&b.path));
    Ok(disks)
}

fn human_size(bytes: u64) -> String {
    const UNITS: [&str; 5] = ["B", "KiB", "MiB", "GiB", "TiB"];
    let mut v = bytes as f64;
    let mut i = 0;
    while v >= 1024.0 && i < UNITS.len() - 1 {
        v /= 1024.0;
        i += 1;
    }
    if i == 0 {
        format!("{} {}", bytes, UNITS[i])
    } else {
        format!("{v:.1} {}", UNITS[i])
    }
}

fn json_escape(s: &str) -> String {
    let mut out = String::with_capacity(s.len() + 8);
    for c in s.chars() {
        match c {
            '"' => out.push_str("\\\""),
            '\\' => out.push_str("\\\\"),
            '\n' => out.push_str("\\n"),
            '\r' => out.push_str("\\r"),
            '\t' => out.push_str("\\t"),
            c if c < ' ' => out.push_str(&format!("\\u{:04x}", c as u32)),
            c => out.push(c),
        }
    }
    out
}

// A minimal JSON reader for lsblk -J. Keeping it local avoids a dependency on
// serde in the installer initramfs and keeps the server small.
#[derive(Debug, Clone)]
enum Json {
    Null,
    Bool(bool),
    Num(f64),
    Str(String),
    Arr(Vec<Json>),
    Obj(Vec<(String, Json)>),
}

struct JsonParser<'a> {
    data: &'a [u8],
    pos: usize,
}

impl<'a> JsonParser<'a> {
    fn new(data: &'a [u8]) -> Self {
        Self { data, pos: 0 }
    }
    fn parse(&mut self) -> Result<Json, String> {
        self.ws();
        let v = self.value()?;
        self.ws();
        Ok(v)
    }
    fn ws(&mut self) {
        while self.pos < self.data.len()
            && matches!(self.data[self.pos], b' ' | b'\n' | b'\r' | b'\t')
        {
            self.pos += 1;
        }
    }
    fn value(&mut self) -> Result<Json, String> {
        self.ws();
        match self.data.get(self.pos).copied() {
            Some(b'{') => self.object(),
            Some(b'[') => self.array(),
            Some(b'"') => Ok(Json::Str(self.string()?)),
            Some(b't') => {
                self.literal("true")?;
                Ok(Json::Bool(true))
            }
            Some(b'f') => {
                self.literal("false")?;
                Ok(Json::Bool(false))
            }
            Some(b'n') => {
                self.literal("null")?;
                Ok(Json::Null)
            }
            Some(b'-') | Some(b'0'..=b'9') => self.number(),
            _ => Err(format!("unexpected JSON at {}", self.pos)),
        }
    }
    fn object(&mut self) -> Result<Json, String> {
        self.pos += 1;
        let mut out = Vec::new();
        self.ws();
        if self.data.get(self.pos) == Some(&b'}') {
            self.pos += 1;
            return Ok(Json::Obj(out));
        }
        loop {
            self.ws();
            let key = self.string()?;
            self.ws();
            if self.data.get(self.pos) != Some(&b':') {
                return Err("expected ':'".to_string());
            }
            self.pos += 1;
            let val = self.value()?;
            out.push((key, val));
            self.ws();
            match self.data.get(self.pos) {
                Some(b',') => {
                    self.pos += 1;
                }
                Some(b'}') => {
                    self.pos += 1;
                    break;
                }
                _ => return Err("expected ',' or '}'".to_string()),
            }
        }
        Ok(Json::Obj(out))
    }
    fn array(&mut self) -> Result<Json, String> {
        self.pos += 1;
        let mut out = Vec::new();
        self.ws();
        if self.data.get(self.pos) == Some(&b']') {
            self.pos += 1;
            return Ok(Json::Arr(out));
        }
        loop {
            out.push(self.value()?);
            self.ws();
            match self.data.get(self.pos) {
                Some(b',') => {
                    self.pos += 1;
                }
                Some(b']') => {
                    self.pos += 1;
                    break;
                }
                _ => return Err("expected ',' or ']'".to_string()),
            }
        }
        Ok(Json::Arr(out))
    }
    fn string(&mut self) -> Result<String, String> {
        if self.data.get(self.pos) != Some(&b'"') {
            return Err("expected string".to_string());
        }
        self.pos += 1;
        let mut out = String::new();
        while let Some(&b) = self.data.get(self.pos) {
            self.pos += 1;
            match b {
                b'"' => return Ok(out),
                b'\\' => {
                    let esc = *self.data.get(self.pos).ok_or("truncated escape")?;
                    self.pos += 1;
                    match esc {
                        b'"' => out.push('"'),
                        b'\\' => out.push('\\'),
                        b'/' => out.push('/'),
                        b'b' => out.push('\u{0008}'),
                        b'f' => out.push('\u{000c}'),
                        b'n' => out.push('\n'),
                        b'r' => out.push('\r'),
                        b't' => out.push('\t'),
                        b'u' => {
                            let h = self
                                .data
                                .get(self.pos..self.pos + 4)
                                .ok_or("truncated unicode escape")?;
                            let h = String::from_utf8_lossy(h);
                            let n =
                                u32::from_str_radix(&h, 16).map_err(|_| "bad unicode escape")?;
                            self.pos += 4;
                            out.push(char::from_u32(n).unwrap_or('\u{fffd}'));
                        }
                        _ => return Err("bad escape".to_string()),
                    }
                }
                b if b < 0x80 => out.push(b as char),
                _ => {
                    // Reconstruct the remaining UTF-8 sequence.
                    let start = self.pos - 1;
                    let width = if b & 0xe0 == 0xc0 {
                        2
                    } else if b & 0xf0 == 0xe0 {
                        3
                    } else {
                        4
                    };
                    let end = (start + width).min(self.data.len());
                    out.push_str(&String::from_utf8_lossy(&self.data[start..end]));
                    self.pos = end;
                }
            }
        }
        Err("unterminated string".to_string())
    }
    fn number(&mut self) -> Result<Json, String> {
        let start = self.pos;
        while self.pos < self.data.len()
            && matches!(
                self.data[self.pos],
                b'-' | b'+' | b'.' | b'e' | b'E' | b'0'..=b'9'
            )
        {
            self.pos += 1;
        }
        let s = String::from_utf8_lossy(&self.data[start..self.pos]);
        s.parse::<f64>()
            .map(Json::Num)
            .map_err(|_| format!("bad number {s}"))
    }
    fn literal(&mut self, s: &str) -> Result<(), String> {
        if self.data[self.pos..].starts_with(s.as_bytes()) {
            self.pos += s.len();
            Ok(())
        } else {
            Err("bad literal".to_string())
        }
    }
}

fn json_get<'a>(j: &'a Json, key: &str) -> Option<&'a Json> {
    if let Json::Obj(fields) = j {
        fields.iter().find(|(k, _)| k == key).map(|(_, v)| v)
    } else {
        None
    }
}

fn json_str(j: &Json, key: &str) -> String {
    match json_get(j, key) {
        Some(Json::Str(s)) => s.clone(),
        Some(Json::Null) | None => String::new(),
        Some(Json::Num(n)) => format!("{n}"),
        Some(Json::Bool(b)) => b.to_string(),
        _ => String::new(),
    }
}

fn json_u64(j: &Json, key: &str) -> u64 {
    match json_get(j, key) {
        Some(Json::Num(n)) => *n as u64,
        Some(Json::Str(s)) => s.parse().unwrap_or(0),
        _ => 0,
    }
}

# sur l'installeur graphique Calamares officiel de NixOS, pour que
# l'installation sur disque inclue AUSSI nos services (common.nix), pas
# juste un NixOS générique.
#
# HISTORIQUE / POURQUOI CETTE VERSION (v2) :
# La première version patchait settings.conf avec `sed` (a\ + \n) dans le
# postInstall du dérivé. Testé en réel le 20/09/2026 sur du matériel
# physique (voir la conversation) : le sed n'a PRIS AUCUN EFFET dans le
# bac à sable de build Nix (aucune erreur, mais notre job "shellprocess@
# hitechos" n'apparaissait tout simplement pas dans la séquence exécutée
# par Calamares — silencieusement ignoré). Cause probable : GNU sed
# interprète \n dans le texte d'une commande `a\` différemment selon le
# contexte d'exécution, fragile à travers les sandboxes de build.
#

# complet est écrit tel quel en Nix (chaîne littérale ci-dessous),
# construit à partir du contenu RÉEL de calamares-nixos-extensions au

# 4f9ef79671b — la révision exacte qui a servi à builder l'ISO testée le
# 20/09/2026), avec seulement les deux ajouts nécessaires (une instance
# + une entrée de séquence). Zéro ambiguïté d'échappement shell : soit
# ça compile avec le bon contenu, soit `nix build` échoue directement.
#
# TRANSPARENCE : voir aussi le premier test réel — l'erreur rencontrée
# ("experimental Nix feature flakes is disabled") venait du job STOCK
# "nixos" (nixos-install), pas de ce fichier : corrigée séparément dans
# common.nix (nix.settings.experimental-features). Cette étape
# shellprocess elle-même n'a PAS ENCORE été testée en conditions réelles
# (le premier test a échoué avant d'atteindre ce point de la séquence,
# et de toute façon la version sed ne s'exécutait pas). Prochain test
# réel à surveiller.
#
# Mécanisme (une fois que ça s'exécute) :
# 1. Le job "nixos" (natif Calamares) partitionne, installe un NixOS de
#    base et écrit /etc/nixos/{configuration,hardware-configuration}.nix
#    — on ne touche PAS à ça, ce sont les réglages propres à CE disque
#    précis (bootloader EFI/BIOS, UUIDs de partitions...).
# 2. Notre étape "shellprocess" s'exécute juste après, chrootée dans le
#    système fraîchement installé (dontChroot: false) :
#    a. clone HiTech-OS dans /etc/nixos/hitech-os
#    b. insère l'import de os/common.nix (services, durcissement — AUCUN
#       réglage disque/bootloader dedans, donc pas de conflit avec ce que

#       "./hardware-configuration.nix" du configuration.nix généré
#    c. `nixos-rebuild boot` pour que ce soit actif dès le premier

# HiTech-OS — universal live/target hardware support.
#
# The first physical tests showed that the minimal installer firmware set
# hid Wi-Fi adapters and left audio/GPU support incomplete. This module is
# intentionally shared by the NetInstaller, the Live ISO, and installed
# targets so the same hardware works before and after installation.
{ config, lib, pkgs, ... }:

{
  # Firmware packages such as Broadcom/Intel wireless blobs and some GPU
  # microcode are unfree but redistributable; allow them explicitly.
  nixpkgs.config.allowUnfree = true;

  # Pull the complete linux-firmware + vendor firmware closures, including
  # Wi-Fi, Bluetooth, audio DSP, and GPU firmware.
  hardware.enableAllFirmware = true;
  hardware.enableRedistributableFirmware = true;
  hardware.wirelessRegulatoryDatabase = true;
  hardware.firmware = with pkgs; [
    linux-firmware
    sof-firmware
    alsa-firmware
  ];

  # Load the common wireless/audio/GPU drivers even when udev coldplug would
  # otherwise defer them until after the installer starts.
  boot.kernelModules = [
    # Intel Wi-Fi
    "iwlwifi"
    "iwlmvm"
    # Realtek
    "rtw88_core"
    "rtw88_pci"
    "rtw89_core"
    "rtw89_pci"
    "rtl8xxxu"
    "rtl8188eu"
    "rtl8821ce"
    # MediaTek
    "mt76"
    "mt76x2e"
    "mt7921e"
    "mt7601u"
    # Broadcom / Atheros
    "brcmfmac"
    "brcmsmac"
    "ath10k_pci"
    "ath11k_pci"
    "ath12k"
    # Audio
    "snd_hda_intel"
    "snd_hda_codec_realtek"
    "snd_hda_codec_hdmi"
    "snd_usb_audio"
    # GPU / DRM
    "amdgpu"
    "radeon"
    "i915"
    "xe"
    "nouveau"
    "virtio_gpu"
  ];
  # GPU drivers are deliberately NOT staged into the initrd. Including them
  # pulls every matching firmware blob as well, which produced a 1.44 GiB
  # initrd and a >4 GiB ISO whose initrd ISOLINUX could no longer load on the
  # BIOS path ("Loading initrd...failed: Invalid argument"). They are listed in
  # boot.kernelModules above and are loaded by udev at stage 2, which is early
  # enough for the kiosk/desktop session.

  # common.nix blacklists audio/GPU for the headless patient-zero profile;
  # explicit hardware support wins on installer/live images and desktops.
  boot.blacklistedKernelModules = lib.mkForce [ ];
  hardware.graphics.enable = true;

    enable = true;
    alsa.enable = true;
    alsa.support32Bit = true;
    pulse.enable = true;
    wireplumber.enable = true;
  };

  environment.systemPackages = with pkgs; [
    gparted
    polkit
    pipewire
    wireplumber
    alsa-utils
    pavucontrol
    pciutils
    usbutils
  ];
}

# HiTech-OS — Phase 8 NetInstaller (minimal eframe/egui ISO).
#
# Goal: a small (~800 MB target) live ISO that contains only:
#   - a Wayland kiosk running the Rust `hitech-installer` GUI,
#   - networking tools,
#   - git + curl,
#   - the NixOS installer machinery.
#
# The target closure is NOT in the ISO, but the complete HiTech-OS source tree
# IS embedded at /etc/hitech-os-source. The graphical installer:
#   1. partitions/formats/mounts the target,
#   2. copies the embedded source tree offline-first (git is only a fallback),
#   3. writes the operator's anonymous identity into configuration.nix,
#   4. calls `nixos-install` with the HiTech-Store binary cache as the primary
#      substituter, so the target closure is downloaded instead of rebuilt.
#
# Two outputs are therefore produced by the flake:
#   nixosConfigurations.netinstaller       — this minimal ISO
#   nixosConfigurations.iso                — legacy full graphical ISO
#
# Security contract:
#   - `cachePublicKey` is embedded in the ISO; the cache is trusted only if
#     its signature verifies. Leave it empty in development only.
#   - `allowSourceFallback = false` is the production default once the cache
#     has a stable release; it prevents a compromised/down cache from silently
#     rebuilding unvetted source on the target.

{ config, lib, pkgs, self, hitechosRevision ? "main", ... }:

let
  cfg = config.hitechos.netinstaller;

  # Pull the HiTech-Store/Harmonia cache public key from the option. The NixOS
  # module for the cache server exposes the same value as a read-only option
  # when co-located; this stays a plain string in the ISO so the installer has
  # no dependency on the server module.
  cachePublicKey = cfg.cachePublicKey;

  # Channel-free installer: the ISO does not embed the full Nixpkgs source
  # tree (that alone is hundreds of MB). Instead the live script points
  # NIX_PATH at the exact Nixpkgs revision baked into the ISO; nixos-install
  # fetches it over the network, which the NetInstaller requires anyway for the
  # closure cache. The revision is the same one used to build the ISO, so the
  # installed system stays reproducible.
  nixpkgsArchive =
    if config.system.nixos.revision != null then
      "https://github.com/NixOS/nixpkgs/archive/${config.system.nixos.revision}.tar.gz"
    else
      "https://github.com/NixOS/nixpkgs/archive/nixos-unstable.tar.gz";

  # Graphical HiTech-OS installer (eframe/egui). It replaces Calamares and
  # owns the partition/format/mount/install pipeline.
  hitechInstaller = pkgs.callPackage ../hitech-installer { };

  # Wrapper script that detects missing hardware acceleration and forces
  # software rendering (llvmpipe) so the eframe/glow HUD appears instead of
  # a black screen on VirtualBox/QEMU-without-virgl/headless CI.
  #
  # Detection strategy:
  #  1. If NO DRM render node exists at all (not just renderD0), force
  #     software rendering for both GL and Wayland/Ozone.
  #  2. Check for common virtual-machine indicators (QEMU virtio-gpu, VirtualBox
  #     VBoxGuest, VMware vmware) so llvmpipe is used even when a stub DRM
  #     device is present but hardware acceleration is not usable.
  hitechInstallerKiosk = pkgs.writeScriptBin "hitech-installer-kiosk" ''
    soft=0
    if [ ! -e /dev/dri/renderD0 ] && [ ! -e /dev/dri/card0 ]; then
      soft=1
    fi
    if [ -e /sys/module/virtio_gpu ] || [ -e /sys/module/vboxguest ] || grep -aq vmware /proc/1/environ 2>/dev/null; then
      soft=1
    fi
    if [ "$soft" = "1" ]; then
      export LIBGL_ALWAYS_SOFTWARE=1
      export WLR_RENDERER_ALLOW_SOFTWARE=1
      export WLR_RENDERER=soft
      export EGL_PLATFORM=x11
      export GST_VAAPI_ALL_DRIVERS=1
    fi
    exec ${hitechInstaller}/bin/hitech-installer "$@"
  '';

in
{
  imports = [ ./hardware-universal.nix ];

  options.hitechos.netinstaller = {
    enable = lib.mkOption {
      type = lib.types.bool;
      default = true;
      description = ''
        Enable the Phase 8 NetInstaller ISO configuration. This module is only
        imported by `nixosConfigurations.netinstaller`, so it defaults to true;
        set false only to evaluate the bare upstream minimal ISO.
      '';
    };

    repoUrl = lib.mkOption {
      type = lib.types.str;
      default = "https://github.com/HiTechTN/HiTech-OS.git";
      description = "Git remote fetched by the graphical installer inside the live session.";
    };

    repoRef = lib.mkOption {
      type = lib.types.str;
      default = hitechosRevision;
      defaultText = lib.literalExpression "self.rev or \"main\"";
      description = ''
        Git revision baked into the ISO. In release builds this is the flake's
        `self.rev`, so the installed system is reproducible and auditable.
      '';
    };

    cacheUrl = lib.mkOption {
      type = lib.types.str;
      default = "https://cache.hitechos.local";
      description = "HiTech-Store/Harmonia binary cache base URL.";
    };

    cachePublicKey = lib.mkOption {
      type = lib.types.str;
      default = "hitechos-cache-1:vTmn/dRSOqEFdrpLs13HYOLv1DnYLvYnDfu7icj3dQY=";
      description = ''
        Public key of the binary cache embedded in the ISO. Empty disables
        signature verification; production images MUST set this.
      '';
    };

    nixpkgsArchive = lib.mkOption {
      type = lib.types.str;
      internal = true;
      default = nixpkgsArchive;
      description = ''
        Internal: Nixpkgs tarball URL exported as NIX_PATH by the graphical
        installer. Derived from the ISO's exact Nixpkgs revision.
      '';
    };


    allowSourceFallback = lib.mkOption {
      type = lib.types.bool;
      default = true;
      description = ''
        If true, the installer continues with a source build when the cache is
        unreachable. Set false for production/release images so a cache outage
        fails closed instead of silently rebuilding unvetted inputs.
      '';
    };
  };

  config = lib.mkIf cfg.enable {
    # --- Base live system -------------------------------------------------
    # Offline-first installer source bundle. `self` is the flake source
    # (tracked files only), so the live ISO carries the exact revision that
    # built it without .git, result symlinks, or local artifacts.
    environment.etc."hitech-os-source".source = self;

    networking.hostName = "hitechos-netinstaller";
    networking.networkmanager.enable = true;

    nix.settings = {
      experimental-features = [ "nix-command" "flakes" ];
      # Prefer the HiTech cache; cache.nixos.org remains the upstream fallback
      # contributed by the NixOS module defaults.
      substituters = lib.mkBefore [ cfg.cacheUrl ];
      trusted-public-keys = lib.mkAfter (lib.optional (cachePublicKey != "") cachePublicKey);
      connect-timeout = 10;
      fallback = true;
    };

    # --- Graphical installer surface -------------------------------------
    # cage is a tiny Wayland kiosk compositor. The eframe/egui installer owns
    # the whole installation flow, so Calamares/Qt/KDE are no longer pulled in.
    services.cage = {
      enable = true;
      user = "nixos";
      program = "${hitechInstallerKiosk}/bin/hitech-installer-kiosk";
      environment = {
        # Consumed by the Rust installer at install time.
        HITECH_CACHE_URL = cfg.cacheUrl;
        HITECH_CACHE_KEY = cachePublicKey;
        HITECH_NIXPKGS_ARCHIVE = nixpkgsArchive;
        HITECH_STATE_VERSION = config.system.stateVersion;
        HITECH_REPO_URL = cfg.repoUrl;
        HITECH_REPO_REF = cfg.repoRef;
        HITECH_ALLOW_SOURCE_FALLBACK = lib.boolToString cfg.allowSourceFallback;
        # eframe/glow: prefer the Wayland/EGL path inside Cage.
        WINIT_UNIX_BACKEND = "wayland";
        # Default to allowing software fallbacks; the kiosk wrapper will force
        # them when no usable DRM render node is detected at launch time.
        WLR_NO_HARDWARE_CURSORS = "1";
        WLR_RENDERER_ALLOW_SOFTWARE = "1";
        WLR_RENDERER = "soft";
        LIBGL_ALWAYS_SOFTWARE = "1";
        XDG_RUNTIME_DIR = "/run/user/1000";
      };
    };

    services.libinput.enable = true;
    # eframe/glow needs EGL/GL. The minimal installation-cd base does not set
    # this, so enable Mesa/DRM explicitly for the kiosk compositor. amdgpu and
    # virtio_gpu are loaded early so both the RX 580 and QEMU virtio-vga have a
    # DRM device by the time Cage starts.
    hardware.graphics.enable = true;
    # GPU modules/firmware are provided by hardware-universal.nix.
    security.polkit.enable = true;

    # The GUI runs as the unprivileged live `nixos` user and escalates only the
    # installer commands through sudo. This NOPASSWD policy is scoped to the
    # ephemeral live ISO and must never be copied to the target.
    security.sudo.wheelNeedsPassword = false;

    # Do NOT use lib.mkForce here. The installation-cd base contributes the
    # login/tty/shell packages; overriding the list is what removed bash and
    # caused the black screen/login loop. Merge our live tools with the base
    # set and make bashInteractive/coreutils explicit as a safety net.
    environment.systemPackages = with pkgs; [
      bashInteractive
      coreutils
      util-linux
      iproute2
      networkmanager
      parted
      gptfdisk
      dosfstools
      e2fsprogs
      btrfs-progs
      nixos-install-tools
      git
      curl
      foot
      mesa
      libglvnd
      hitechInstaller
      pkgs.which
      # Kept for the installer pipeline and live debugging.
      sudo
      nix
      efibootmgr
      efivar
      kmod
      hwdata
      pciutils
      usbutils
    ];

    # --- Deliberate size reductions --------------------------------------
    # The stock installation-cd base is built for every possible machine:
    # full Nixpkgs channel, all firmware, ZFS, software RAID, OpenSSH. The
    # NetInstaller needs none of that to reach the cache; the closures it
    # installs come from HiTech-Store. Wired Ethernet (and a curated set of
    # initrd storage/USB modules) is the supported live networking path.
    system.installer.channel.enable = false;
    system.extraDependencies = lib.mkForce [ ];
    # hardware-enableAllFirmware / vendor blobs are enabled by
    # hardware-universal.nix; do not force them off here.
    boot.swraid.enable = lib.mkForce false;
    services.openssh.enable = lib.mkForce false;
    boot.supportedFilesystems.zfs = lib.mkForce false;

    # Minimal initrd module set: enough for IDE/SATA/NVMe/USB/virtio disks
    # and the squashfs live root, without pulling all-hardware firmware.
    boot.initrd.availableKernelModules = [
      "ahci"
      "ata_piix"
      "nvme"
      "uas"
      "usb_storage"
      "xhci_pci"
      "ehci_pci"
      "ohci_pci"
      "usbhid"
      "hid_generic"
      "sdhci"
      "sdhci_pci"
      "virtio_blk"
      "virtio_pci"
      "virtio_scsi"
      "virtio_net"
      "ext4"
      "vfat"
      "squashfs"
      "overlay"
      "loop"
    ];

    # Keep the installation-cd base default packages (installation-cd-base
    # sets this to [ rsync ]); do NOT force it empty. bash/coreutils are also
    # explicit systemPackages above.
    programs.command-not-found.enable = lib.mkForce false;
    xdg.autostart.enable = lib.mkForce false;
    xdg.icons.enable = lib.mkForce false;
    xdg.mime.enable = lib.mkForce false;
    xdg.sounds.enable = lib.mkForce false;
    services.logrotate.enable = lib.mkForce false;
    services.udisks2.enable = lib.mkForce false;

    documentation.enable = lib.mkForce false;
    documentation.man.enable = lib.mkForce false;
    documentation.nixos.enable = lib.mkForce false;
    i18n.supportedLocales = [
      "en_US.UTF-8/UTF-8"
      "fr_FR.UTF-8/UTF-8"
    ];
    i18n.defaultLocale = "en_US.UTF-8";
    console.keyMap = "us";
    fonts.packages = [ pkgs.dejavu_fonts ];

    # --- Live credentials (ephemeral; never reused on the target) ---------
    # No hardcoded live password: the kiosk session is passwordless by design
    # and the operator identity is created exclusively by the GUI form.
    users.users.nixos.initialHashedPassword = lib.mkForce "";
    users.users.nixos.extraGroups = [
      "wheel"
      "networkmanager"
      "video"
      "render"
      "input"
      "tty"
    ];

    # Always keep a usable shell reachable: getty@tty2/tty3 autologin as the
    # live user, and ttyS0 gets an explicit getty for serial/headless rescue.
    services.getty.autologinUser = "nixos";
    systemd.services."serial-getty@ttyS0" = {
      enable = true;
      wantedBy = [ "getty.target" ];
      environment.TERM = "linux";
      serviceConfig.ExecStart = [
        ""
        "${pkgs.util-linux}/bin/agetty --autologin nixos --login-program ${pkgs.shadow}/bin/login --noclear --keep-baud ttyS0 115200,38400,9600 linux"
      ];
    };

    # eframe/egui + Mesa are much smaller than the old Calamares/Qt/KDE stack,
    # but xz still compresses this closure noticeably better than zstd-19 at
    # the cost of a slower image build; the compressed ISO is what users get.
    isoImage.squashfsCompression = lib.mkDefault "xz -Xdict-size 100%";

    # --- Boot media identity ---------------------------------------------
    isoImage.volumeID = "HITECHOS_NET";
    isoImage.edition = "netinstaller";
    image.fileName = lib.mkForce
      "hitechos-netinstaller-${config.system.nixos.label}-${pkgs.stdenv.hostPlatform.system}.iso";
  };
}

# HiTech-OS — service natif « genio » (daemon local + dashboard + CLI)
#
# Remplace le lancement manuel `python -m genio_server.daemon` du lab :
#   - package reproductible (apps/genio/default.nix) ;
#   - état persistant dans /var/lib/genio (mémoire, audit, fichiers) ;
#   - clé de capacité générée une seule fois au premier boot ;
#   - Ollama local démarré et modèle préchargé (si pullModel = true) ;
#   - dépendance souple à hitech-store (le dashboard continue de démarrer
#     même si le store est indisponible, avec dégradation explicite).
#
# Le daemon n'écoute QUE sur la loopback par défaut : le dashboard n'a pas
# d'authentification utilisateur. Pour y accéder à distance, utiliser un
# tunnel SSH (`ssh -L 8765:127.0.0.1:8765 ...`) ou activer openFirewall en
# connaissance de cause.

{ config, lib, pkgs, ... }:

let
  cfg = config.services.genio;
  genio = pkgs.callPackage ../../apps/genio/default.nix { };

  keygen = pkgs.writeShellScript "genio-keygen" ''
    set -eu
    key=${lib.escapeShellArg "${cfg.stateDir}/cap.key"}
    if [ ! -s "$key" ]; then
      umask 077
      ${pkgs.openssl}/bin/openssl rand -hex 32 > "$key"
      chmod 0600 "$key"
    fi
  '';
in
{
  options.services.genio = {
    enable = lib.mkOption {
      type = lib.types.bool;
      default = true;
      description = "Enable the Genio daemon, dashboard and CLI.";
    };

    host = lib.mkOption {
      type = lib.types.str;
      default = "127.0.0.1";
      description = "Address the Genio HTTP daemon listens on.";
    };

    port = lib.mkOption {
      type = lib.types.port;
      default = 8765;
      description = "Genio HTTP port (dashboard is served at /ui/).";
    };

    stateDir = lib.mkOption {
      type = lib.types.str;
      default = "/var/lib/genio";
      description = "Persistent Genio state directory (memory DB, audit, files).";
    };

    model = lib.mkOption {
      type = lib.types.str;
      default = "qwen2.5:7b-instruct";
      description = "Ollama model used by the local Genio backend.";
    };

    ollama.enable = lib.mkOption {
      type = lib.types.bool;
      default = true;
      description = "Enable the local Ollama service for Genio.";
    };

    ollama.pullModel = lib.mkOption {
      type = lib.types.bool;
      default = true;
      description = ''
        Pull `services.genio.model` into Ollama on first boot. Model weights
        are downloaded over the network and are not part of the Nix closure.
      '';
    };

    openFirewall = lib.mkOption {
      type = lib.types.bool;
      default = false;
      description = "Open the Genio port in the NixOS firewall (not recommended: no user auth).";
    };
  };

  config = lib.mkIf cfg.enable {
    users.users.genio = {
      isSystemUser = true;
      group = "genio";
      home = cfg.stateDir;
      createHome = false;
    };
    users.groups.genio = { };

    systemd.tmpfiles.rules = [
      "d ${cfg.stateDir} 0750 genio genio -"

# HiTech-OS — binary cache for the NetInstaller and OTA closures.
#
# Design rationale (see docs/audit/PHASE8-NETINSTALLER-RUST-DE.md):
#   - Harmonia is the primary cache server: Rust, low overhead, multi-worker,
#     and it signs store paths with a dedicated key.
#   - nix-serve is documented as a fallback if a host already uses it, but it
#     does not sign and is single-process; this module intentionally uses
#     Harmonia for the new cache.
#   - The cache is write-protected and read-only to clients. CI never gets a
#     private key in a writable repository: it builds closures and pushes them
#     to the cache host over SSH; the cache host signs them.
#   - TLS terminates in nginx with ACME. Plain HTTP is only for the loopback
#     listener.
#
# Deployment contract:
#   /run/secrets/hitech-cache-signing-key — secret key generated offline with
#       nix-store --generate-binary-cache-key hitechos-cache-1 <secret> <public>
#   services.hitech-cache.publicKey       — the matching public key, embedded
#       in clients and in the NetInstaller ISO.
#
# IMPORTANT: keep the private key out of Git. The public key alone is safe to
# commit and must be embedded in the installer so cache signatures can be
# verified.

{ config, lib, pkgs, ... }:

let
  cfg = config.services.hitech-cache;
in
{
  options.services.hitech-cache = {
    enable = lib.mkEnableOption "Harmonia binary cache server for HiTech-OS closures";

    package = lib.mkPackageOption pkgs "harmonia" { };

    bindAddress = lib.mkOption {
      type = lib.types.str;
      default = "127.0.0.1";
      description = "Address Harmonia binds to; nginx proxies external traffic.";
    };

    port = lib.mkOption {
      type = lib.types.port;
      default = 5000;
      description = "Internal Harmonia TCP port.";
    };

    domain = lib.mkOption {
      type = lib.types.str;
      default = "cache.hitechos.local";
      description = "Public hostname served by nginx (TLS/ACME).";
    };

    enableTls = lib.mkOption {
      type = lib.types.bool;
      default = false;
      description = ''
        Terminate TLS with nginx + ACME. Enable once `domain` resolves to this
        host and port 80/443 are reachable from the internet. Disable only for
        a private network/VPN cache.
      '';
    };

    signKeyPath = lib.mkOption {
      type = lib.types.path;
      default = "/run/secrets/hitech-cache-signing-key";
      description = ''
        Path to the secret signing key. Provision it with sops/agenix, never
        with systemd-tmpfiles and never in the Nix store.
      '';
    };

    publicKey = lib.mkOption {
      type = lib.types.str;
      default = "";
      example = "hitechos-cache-1:AbCdEf...=";
      description = ''
        Public key corresponding to `signKeyPath`. Required when the server is
        enabled; clients must trust this key to accept HiTech-OS closures.
      '';
    };

    openFirewall = lib.mkOption {
      type = lib.types.bool;
      default = false;
      description = "Open the public cache ports in the NixOS firewall.";
    };

    client = {
      enable = lib.mkOption {
        type = lib.types.bool;
        default = false;
        description = ''
          Configure this host to substitute from the HiTech-OS cache. Enable on
          patient-zero, installed targets and the NetInstaller live system.
        '';
      };

      url = lib.mkOption {
        type = lib.types.str;
        default = "https://cache.hitechos.local";
        description = "Base URL of the HiTech-OS binary cache.";
      };

      publicKey = lib.mkOption {
        type = lib.types.str;
        default = "";
        description = "Trusted public key for `url` (must match the server).";
      };
    };
  };

  config = lib.mkMerge [
    (lib.mkIf cfg.enable {
      assertions = [
        {
          assertion = cfg.publicKey != "";
          message = ''
            services.hitech-cache.enable requires services.hitech-cache.publicKey
            so clients can verify cache signatures.
          '';
        }
      ];

      services.harmonia = {
        inherit (cfg) package;

        cache = {
          enable = true;
          signKeyPaths = [ cfg.signKeyPath ];
          settings.bind = "${cfg.bindAddress}:${toString cfg.port}";
        };
      };

      services.nginx = lib.mkIf cfg.enableTls {
        enable = true;
        recommendedProxySettings = true;
        recommendedTlsSettings = true;

        virtualHosts.${cfg.domain} = {
          enableACME = true;
          forceSSL = true;
          locations."/".proxyPass = "http://${cfg.bindAddress}:${toString cfg.port}";
        };
      };

      security.acme.acceptTerms = lib.mkIf cfg.enableTls true;

      networking.firewall.allowedTCPPorts = lib.optionals cfg.openFirewall (
        if cfg.enableTls then

        else
          [ cfg.port ]

    })

    (lib.mkIf cfg.client.enable {
      nix.settings = {
        substituters = lib.mkAfter [ cfg.client.url ];
        trusted-public-keys = lib.mkAfter (

# HiTech-OS — service natif « hitech-store » (portail universel FastAPI)
#
# Packagé via la derivation « à la app2nix » : apps/hitech-store/default.nix.
# Importé par os/installation.nix, donc présent sur patient-zero ET sur
# toute installation Calamares (active par défaut, désactivable).
#
# Secrets : /run/secrets/hitech-store.env est un placeholder créé par
# systemd-tmpfiles UNIQUEMENT s'il n'existe pas. os/services/secrets.nix le
# remplace avant le démarrage si /etc/hitechos/hitech-store.env.enc existe.
# Le panel admin reste utilisable sans SMTP en mode DEV_OTP_FALLBACK=1
# (PIN écrit dans le journal) ; passer devOtpFallback=false en production.

{ config, lib, pkgs, ... }:

let
  cfg = config.services.hitech-store;
  hitechStore = pkgs.callPackage ../../apps/hitech-store/default.nix { };

  # The tmpfiles placeholder ships a publicly known SECRET_KEY so that the
  # service can boot before sops secrets are provisioned. That key signs the
  # admin JWT: if it survives into an exposed deployment, anyone can forge an
  # admin session. This oneshot replaces it with a per-boot random key when the
  # placeholder (or an empty value) is still present; a real sops-provisioned
  # key is never touched. Runs after hitechos-secrets, before the service.
  envKeygen = pkgs.writeShellScript "hitech-store-keygen" ''
    set -eu
    umask 077
    env_file=${lib.escapeShellArg cfg.environmentFile}
    [ -f "$env_file" ] || exit 0

    placeholder='^SECRET_KEY=dev-only-placeholder-change-me$'
    if ${pkgs.gnugrep}/bin/grep -q "$placeholder" "$env_file" \
      || ! ${pkgs.gnugrep}/bin/grep -q '^SECRET_KEY=.' "$env_file"; then
      new_key="$(${pkgs.openssl}/bin/openssl rand -hex 32)"
      tmp="$env_file.tmp.$$"
      trap '${pkgs.coreutils}/bin/rm -f "$tmp"' EXIT
      ${pkgs.gawk}/bin/awk -v key="$new_key" '
        BEGIN { replaced = 0 }
        /^SECRET_KEY=/ { print "SECRET_KEY=" key; replaced = 1; next }
        { print }
        END { if (!replaced) print "SECRET_KEY=" key }
      ' "$env_file" > "$tmp"
      ${pkgs.coreutils}/bin/chmod --reference="$env_file" "$tmp" 2>/dev/null || true
      ${pkgs.coreutils}/bin/chown --reference="$env_file" "$tmp" 2>/dev/null || true
      ${pkgs.coreutils}/bin/mv -f "$tmp" "$env_file"
      trap - EXIT
      echo "hitech-store-keygen: generated a per-boot SECRET_KEY (placeholder was still present)" >&2
    fi
  '';
in

# HiTech-OS — déchiffrement sops des secrets de service au boot.
#
# Pourquoi un oneshot et pas sops-nix comme module NixOS ?
# L'installateur Calamares génère un /etc/nixos/configuration.nix NON-flake.
# Un `imports = [ sops-nix.nixosModules.sops ]` y serait impossible sans
# flake/input disponible au moment du `nixos-rebuild` local. On garde donc
# le contrat NixOS simple : des fichiers chiffrés déposés hors dépôt, et
# sops (paquet) appelé au démarrage.
#
# Contrat de déploiement (répertoire root:root 0700) :
#   /etc/hitechos/age.key                  clé age de l'hôte (jamais commitée)
#   /etc/hitechos/hitech-store.env.enc     secrets du portail (format dotenv)
#   /etc/hitechos/genio.env.enc            tokens store/OpenRouter de Genio
#
# Création (exemple) :
#   sops --encrypt --age "$(cat /etc/hitechos/age.pub)" \
#        --input-type dotenv --output-type dotenv \
#        /etc/hitechos/hitech-store.env > /etc/hitechos/hitech-store.env.enc
#
# Sans fichiers .enc, le service ne fait rien : les placeholders tmpfiles
# restent en place et les services démarrent en mode dégradé explicite.

{ config, lib, pkgs, ... }:

let
  secretsDir = "/etc/hitechos";
  decrypt = pkgs.writeShellScript "hitechos-decrypt-secrets" ''
    set -eu
    umask 077
    ${pkgs.coreutils}/bin/install -d -m 0700 /run/secrets

    group_exists() {
      while IFS=: read -r name _; do
        [ "$name" = "$1" ] && return 0
      done < /etc/group
      return 1
    }

    decrypt_one() {
      src="$1"
      dst="$2"
      group="$3"
      [ -f "$src" ] || return 0
      if [ ! -r "${secretsDir}/age.key" ]; then
        echo "hitechos-secrets: $src exists but ${secretsDir}/age.key is missing" >&2
        return 1
      fi
      if ! group_exists "$group"; then
        echo "hitechos-secrets: group $group absent, skipping $dst" >&2
        return 0
      fi
      tmp="$(${pkgs.coreutils}/bin/mktemp)"
      trap '${pkgs.coreutils}/bin/rm -f "$tmp"' EXIT
      SOPS_AGE_KEY_FILE="${secretsDir}/age.key" \
        ${pkgs.sops}/bin/sops --decrypt "$src" > "$tmp"
      ${pkgs.coreutils}/bin/install -o root -g "$group" -m 0640 "$tmp" "$dst"
      ${pkgs.coreutils}/bin/rm -f "$tmp"
      trap - EXIT
    }

    decrypt_one "${secretsDir}/hitech-store.env.enc" /run/secrets/hitech-store.env hitech-store
    decrypt_one "${secretsDir}/genio.env.enc" /run/secrets/genio.env genio
  '';
in
{
  environment.etc."hitechos/README".text = ''
    HiTech-OS secrets
    =================
    Place the age private key in /etc/hitechos/age.key (mode 0600).
    Encrypt dotenv files with sops and name them:
      hitech-store.env.enc
      genio.env.enc
    They are decrypted to /run/secrets/ before the services start.
    Required hitech-store keys: SECRET_KEY, SMTP_HOST, SMTP_PORT, SMTP_USER,
      SMTP_PASSWORD, AGENT_SHARED_SECRET, STORAGE_DIR.
    Optional GitHub OAuth keys: GITHUB_CLIENT_ID, GITHUB_CLIENT_SECRET,
      GITHUB_ALLOWED_LOGIN, GITHUB_REDIRECT_URI.
    Required genio keys: HITECH_STORE_ADMIN_TOKEN, HITECH_STORE_AGENT_TOKEN;
      optional GENIO_OPENROUTER_KEY / OPENROUTER_API_KEY.
  '';

  systemd.services.hitechos-secrets = {
    description = "Decrypt HiTech-OS service secrets with sops";
    wantedBy = [ "multi-user.target" ];
    before = [ "hitech-store.service" "genio.service" ];
    serviceConfig = {
      Type = "oneshot";
      RemainAfterExit = true;
      ExecStart = "${decrypt}";
    };
  };
}

#!/usr/bin/env bash
# Verify that a HiTech-OS ISO carries the expected hitech-installer payload.
#
# Usage:
#   ./scripts/e2e/check-iso-contents.sh NETINSTALLER.iso LIVE.iso [INSTALLER_STORE_BASENAME]
#
# The optional third argument is the expected store basename, for example:
#   i5g7srfdxxfgy3qmlwmmjigs4rb2b4sp-hitech-installer-2.0.0
#
# Mounting requires root; the script re-executes mount/umount through sudo when
# it is not already root.
set -euo pipefail

usage() {
  sed -n '2,12p' "$0" | sed 's/^# \{0,1\}//'
}

if [[ $# -lt 2 || $# -gt 3 ]]; then
  usage >&2
  exit 2
fi

net_iso=$1
live_iso=$2
expected=${3:-}

for iso in "$net_iso" "$live_iso"; do
  [[ -f "$iso" ]] || { echo "missing ISO: $iso" >&2; exit 1; }
done

if [[ $(id -u) -eq 0 ]]; then
  as_root() { "$@"; }
else
  as_root() { sudo -- "$@"; }
fi

work=$(mktemp -d)
iso_mnt="$work/iso"
store_mnt="$work/store"
mkdir -p "$iso_mnt" "$store_mnt"
iso_mounted=0
store_mounted=0

cleanup() {
  if [[ $store_mounted -eq 1 ]]; then as_root umount "$store_mnt" || true; fi
  if [[ $iso_mounted -eq 1 ]]; then as_root umount "$iso_mnt" || true; fi
  rm -rf "$work"
}
trap cleanup EXIT

check_iso() {
  local label=$1 iso=$2 squash
  echo "== $label: $iso"
  as_root mount -o loop,ro "$iso" "$iso_mnt"
  iso_mounted=1

  squash=$(find "$iso_mnt" -maxdepth 1 -name 'nix-store.squashfs' -print -quit)
  if [[ -z "$squash" ]]; then
    echo "  no nix-store.squashfs found" >&2
    return 1
  fi

  as_root mount -o loop,ro "$squash" "$store_mnt"
  store_mounted=1

  local found
  found=$(find "$store_mnt" -maxdepth 1 -type d -name '*hitech-installer*' -printf '%f\n' | sort)
  if [[ -z "$found" ]]; then
    echo "  no hitech-installer store path found" >&2
    return 1
  fi
  echo "$found" | sed 's/^/  /'

  if [[ -n "$expected" ]] && ! grep -Fxq "$expected" <<<"$found"; then
    echo "  expected payload not found: $expected" >&2
    return 1
  fi

  as_root umount "$store_mnt"
  store_mounted=0
  as_root umount "$iso_mnt"
  iso_mounted=0
}

check_iso NetInstaller "$net_iso"
check_iso Live "$live_iso"

echo "ISO payload check: PASS"

#!/usr/bin/env bash
#
# HiTech-OS — publish system closures to the Harmonia binary cache.
#
# Builds one or more flake targets, optionally signs them with a local secret
# key, uploads the closure to the cache host over SSH, and optionally lets the
# cache host sign the closure with the key that lives there
# (default: /run/secrets/hitechos-cache-signing-key).
#
# Security model (see docs/audit/PHASE8-NETINSTALLER-RUST-DE.md §5):
#   - The cache host owns the private signing key. CI never receives it.
#   - CI copies closures over a restricted SSH key, then the cache host signs
#     them. Clients verify the signatures with the matching public key.
#   - A trusted builder that does have the key can sign locally with --sign-key.
#
# Usage:
#   scripts/publish-cache.sh [options] [target ...]
#
# Targets:
#   patient-zero   nixosConfigurations.patient-zero system closure (default)
#   hitech-shell   Rust DE package (default)
#   netinstaller   NetInstaller ISO
#   iso            legacy full ISO
#   <installable>  any flake installable passed through to `nix build`
#
# Options:
#   --repo PATH          repository root (default: git toplevel)
#   --host user@host     cache host SSH target (or HITECH_CACHE_HOST)
#   --sign-key PATH      local secret key; signs before upload
#   --remote-key PATH    key path on the cache host; signs after upload
#   --targets "a b"      space-separated targets (default: patient-zero hitech-shell)
#   --no-push            build (and optionally sign) locally, skip the network
#   --no-sign            never sign, even if a key was supplied
#   --dry-run            print what would be built/pushed
#   -h, --help           show this help
#
# Environment:
#   HITECH_CACHE_HOST         cache host SSH target ("user@host")
#   HITECH_CACHE_SIGNING_KEY  local secret key (trusted builders only)
#   HITECH_CACHE_REMOTE_KEY   key path on the cache host
#   HITECH_CACHE_TARGETS      space-separated default targets
#   HITECH_REPO_ROOT          override repository root
#
# Examples:
#   # trusted builder with the key mounted locally
#   scripts/publish-cache.sh --sign-key /secure/hitechos-cache-1.secret
#
#   # CI: push unsigned, let the cache host sign
#   HITECH_CACHE_HOST=cache-publisher@cache.hitechos.tn \
#     scripts/publish-cache.sh
#
#   # build and sign locally, no network
#   scripts/publish-cache.sh --no-push --sign-key /secure/hitechos-cache-1.secret

set -euo pipefail

NIX=(nix --extra-experimental-features "nix-command flakes")

die() { printf 'publish-cache: error: %s\n' "$*" >&2; exit 1; }
log() { printf 'publish-cache: %s\n' "$*" >&2; }

usage() { sed -n '2,53p' "$0" | sed 's/^# \{0,1\}//'; }

repo_root() {
  if [ -n "${HITECH_REPO_ROOT:-}" ]; then
    printf '%s' "$HITECH_REPO_ROOT"
    return
  fi
  if root=$(git -C "$(dirname "$0")" rev-parse --show-toplevel 2>/dev/null); then
    printf '%s' "$root"
    return
  fi
  printf '%s' "$(cd "$(dirname "$0")/.." && pwd)"
}

target_attr() {
  case "$1" in
    patient-zero) printf '%s' '.#nixosConfigurations.patient-zero.config.system.build.toplevel' ;;
    hitech-shell) printf '%s' '.#hitech-shell' ;;
    netinstaller) printf '%s' '.#nixosConfigurations.netinstaller.config.system.build.isoImage' ;;
    iso) printf '%s' '.#nixosConfigurations.iso.config.system.build.isoImage' ;;
    *) printf '%s' "$1" ;;
  esac
}

# Sign every path in the closure. Nix verifies each substituted path, so every
# closure member needs a signature, not only the toplevel.
sign_closure() {
  local key=$1 path=$2
  [ -r "$key" ] || die "signing key is not readable: $key"
  log "signing closure of $path with $key"
  "${NIX[@]}" path-info --recursive "$path" | "${NIX[@]}" store sign --key-file "$key" --stdin
}

push_closure() {
  local host=$1 path=$2
  log "pushing $path -> ssh-ng://$host"
  "${NIX[@]}" copy --to "ssh-ng://$host" --no-check-sigs "$path"
}

sign_closure_remote() {
  local host=$1 key=$2 path=$3
  log "signing $path on $host with $key"
  ssh -o BatchMode=yes -o ConnectTimeout=15 "$host" \
    "nix --extra-experimental-features 'nix-command flakes' path-info --recursive '$path' | nix --extra-experimental-features 'nix-command flakes' store sign --key-file '$key' --stdin"
}

verify_remote() {
  local host=$1 path=$2
  "${NIX[@]}" path-info --store "ssh-ng://$host" "$path" >/dev/null
  log "verified present in cache: $path"
}

main() {
  local repo="" host="" sign_key="" remote_key="" no_push=0 no_sign=0 dry_run=0
  local -a targets=()

  while [ $# -gt 0 ]; do
    case "$1" in
      --repo) repo=$2; shift 2 ;;
      --host) host=$2; shift 2 ;;
      --sign-key) sign_key=$2; shift 2 ;;
      --remote-key) remote_key=$2; shift 2 ;;
      --targets) read -r -a targets <<< "$2"; shift 2 ;;
      --no-push) no_push=1; shift ;;
      --no-sign) no_sign=1; shift ;;
      --dry-run) dry_run=1; shift ;;
      -h|--help) usage; exit 0 ;;
      --) shift; while [ $# -gt 0 ]; do targets+=("$1"); shift; done ;;
      -*) die "unknown option: $1" ;;
      *) targets+=("$1"); shift ;;
    esac
  done

  repo=${repo:-$(repo_root)}
  host=${host:-${HITECH_CACHE_HOST:-}}
  sign_key=${sign_key:-${HITECH_CACHE_SIGNING_KEY:-}}
  remote_key=${remote_key:-${HITECH_CACHE_REMOTE_KEY:-/run/secrets/hitechos-cache-signing-key}}
  host=${host#ssh-ng://}

  if [ "${#targets[@]}" -eq 0 ]; then
    read -r -a targets <<< "${HITECH_CACHE_TARGETS:-patient-zero hitech-shell}"
  fi

  [ -d "$repo" ] || die "repository root not found: $repo"
  command -v nix >/dev/null || die "nix is not in PATH"

  if [ "$no_push" -eq 0 ] && [ "$dry_run" -eq 0 ] && [ -z "$host" ]; then
    die "no cache host: pass --host user@host or set HITECH_CACHE_HOST (or use --no-push)"
  fi

  log "repository: $repo"
  log "targets:    ${targets[*]}"
  [ -n "$host" ] && log "cache host: $host"
  [ "$no_push" -eq 1 ] && log "mode:       build only (--no-push)"
  [ "$no_sign" -eq 1 ] && log "mode:       signing disabled (--no-sign)"

  cd "$repo"

  local name attr out
  for name in "${targets[@]}"; do
    attr=$(target_attr "$name")
    log "building $name ($attr)"

    if [ "$dry_run" -eq 1 ]; then
      printf '  nix build %s --no-link --print-out-paths\n' "$attr"
      continue
    fi

    out=$("${NIX[@]}" build "$attr" --no-link --print-out-paths -L)
    [ -n "$out" ] || die "build produced no output for $name"
    log "built $out"

    if [ "$no_sign" -eq 0 ] && [ -n "$sign_key" ]; then
      sign_closure "$sign_key" "$out"
    fi

    if [ "$no_push" -eq 0 ]; then
      push_closure "$host" "$out"
      if [ "$no_sign" -eq 0 ] && [ -z "$sign_key" ]; then
        sign_closure_remote "$host" "$remote_key" "$out"
      fi
      verify_remote "$host" "$out"
    fi
  done

  log "done"
}

main "$@"

#!/usr/bin/env bash
#
# HiTech-OS — sign a store path and its closure with the cache signing key.
#
# Runs on the cache host (or any trusted signer that owns the private key).
# Intended to be called over SSH by scripts/publish-cache.sh and the
# .github/workflows/cache-publish.yml pipeline; it can also be used manually
# after a closure has been copied into the cache host's store.
#
# Usage:
#   scripts/sign-cache-closure.sh [-k KEY] PATH [PATH ...]
#
# Options:
#   -k, --key PATH   secret signing key (default: $HITECH_CACHE_SIGNING_KEY or
#                    /run/secrets/hitechos-cache-signing-key)
#   -h, --help       show this help
#
# Environment:
#   HITECH_CACHE_SIGNING_KEY  default key path
#
# Example (on the cache host):
#   scripts/sign-cache-closure.sh /nix/store/<hash>-nixos-system-patient-zero-<ver>

set -euo pipefail

NIX=(nix --extra-experimental-features "nix-command flakes")

die() { printf 'sign-cache-closure: error: %s\n' "$*" >&2; exit 1; }
log() { printf 'sign-cache-closure: %s\n' "$*" >&2; }

usage() { sed -n '2,22p' "$0" | sed 's/^# \{0,1\}//'; }

key=${HITECH_CACHE_SIGNING_KEY:-/run/secrets/hitechos-cache-signing-key}
paths=()

while [ $# -gt 0 ]; do
  case "$1" in
    -k|--key) key=$2; shift 2 ;;
    -h|--help) usage; exit 0 ;;
    --) shift; while [ $# -gt 0 ]; do paths+=("$1"); shift; done ;;
    -*) die "unknown option: $1" ;;
    *) paths+=("$1"); shift ;;
  esac
done

[ "${#paths[@]}" -gt 0 ] || { usage; exit 2; }
[ -r "$key" ] || die "signing key is not readable: $key"
command -v nix >/dev/null || die "nix is not in PATH"

for path in "${paths[@]}"; do
  [ -e "$path" ] || die "store path does not exist: $path"
  log "signing closure of $path"
  # Sign every closure member: Nix verifies each path it substitutes.
  "${NIX[@]}" path-info --recursive "$path" | "${NIX[@]}" store sign --key-file "$key" --stdin
  log "signed: $path"
done

log "done"
